    def get_revenue_forecast(self, days: int = 30) -> Dict[str, Any]:
        daily_avg = self.business_metrics["monthly_revenue"] / 30
        forecast = []

        for i in range(days):
            date = datetime.now() + timedelta(days=i)
            growth_factor = 1 + (0.02 * (i / 30))
            projected_revenue = daily_avg * growth_factor * (0.8 + 0.4 * (i % 7) / 7)

            forecast.append({
                "date": date.strftime("%Y-%m-%d"),
                "projected_revenue": round(projected_revenue, 2),
                "confidence": max(90 - (i * 1.5), 50)
            })

        return {
            "forecast": forecast,
            "total_projected": sum(f["projected_revenue"] for f in forecast),
            "avg_confidence": sum(f["confidence"] for f in forecast) / len(forecast)
        }
