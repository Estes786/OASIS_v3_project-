class OasisContentGenerator:
    """
    OASIS Content Generation Engine - Ultra-lightweight API-first implementation
    Zero heavy dependencies, 100% Hugging Face API powered
    """

    def __init__(self):
        self.hf_token = os.getenv("HF_TOKEN", "")
        self.api_base = "https://api-inference.huggingface.co/models"
        self.session_stats = {
            "content_generated": 0,
            "total_tokens": 0,
            "start_time": datetime.now().isoformat(),
            "revenue_generated": 0.0
        }

        # Content templates for different use cases
        self.content_templates = {
            "blog_post": "Write a comprehensive blog post about {topic}. Include an engaging introduction, 3-4 main sections with detailed explanations, and a compelling conclusion. Target audience: {audience}. Tone: {tone}.",
            "social_media": "Create an engaging social media post about {topic}. Make it {tone} and include relevant hashtags. Keep it under 280 characters for Twitter compatibility.",
            "product_description": "Write a compelling product description for {topic}. Highlight key features, benefits, and include a call-to-action. Target audience: {audience}. Tone: {tone}.",
            "email_campaign": "Create an email campaign for {topic}. Include subject line, greeting, main content, and call-to-action. Target audience: {audience}. Tone: {tone}.",
            "article_summary": "Create a concise summary of the following content about {topic}. Focus on key points and main takeaways. Keep it under 200 words.",
            "seo_content": "Write SEO-optimized content about {topic}. Include primary and secondary keywords naturally. Target audience: {audience}. Word count: approximately 500 words."
        }

        # Pricing structure
        self.pricing = {
            "blog_post": {"starter": 25.0, "professional": 50.0, "enterprise": 100.0},
            "social_media": {"starter": 5.0, "professional": 10.0, "enterprise": 20.0}, 
            "product_description": {"starter": 15.0, "professional": 30.0, "enterprise": 60.0},
            "email_campaign": {"starter": 20.0, "professional": 40.0, "enterprise": 80.0},
            "article_summary": {"starter": 10.0, "professional": 20.0, "enterprise": 40.0},
            "seo_content": {"starter": 30.0, "professional": 60.0, "enterprise": 120.0}
        }

    def generate_content(self, content_type: str, topic: str, audience: str = "general", 
                        tone: str = "professional", model: str = "microsoft/DialoGPT-large") -> Dict[str, Any]:
        """
        Generate content using Hugging Face API
        """
        if not topic.strip():
            return {"error": "Please enter a topic"}

        # Get template and format it
        if content_type not in self.content_templates:
            return {"error": f"Unsupported content type: {content_type}"}

        prompt = self.content_templates[content_type].format(
            topic=topic, audience=audience, tone=tone
        )

        # Update session stats
        self.session_stats["content_generated"] += 1

        headers = {"Authorization": f"Bearer {self.hf_token}"}

        try:
            response = requests.post(
                f"{self.api_base}/{model}",
                headers=headers,
                json={
                    "inputs": prompt,
                    "parameters": {
                        "max_length": 512,
                        "temperature": 0.7,
                        "do_sample": True
                    }
                },
                timeout=60
            )

            if response.status_code == 200:
                result = response.json()

                # Process the response
                if isinstance(result, list) and len(result) > 0:
                    generated_text = result[0].get("generated_text", "")

                    # Clean up the generated text
                    generated_content = generated_text.replace(prompt, "").strip()

                    # Calculate pricing
                    pricing = self.pricing.get(content_type, {"starter": 10, "professional": 20, "enterprise": 40})

                    # Update revenue stats (using professional tier for calculation)
                    self.session_stats["revenue_generated"] += pricing["professional"]
                    self.session_stats["total_tokens"] += len(generated_content.split())

                    formatted_result = {
                        "content_type": content_type,
                        "topic": topic,
                        "audience": audience,
                        "tone": tone,
                        "generated_content": generated_content,
                        "word_count": len(generated_content.split()),
                        "character_count": len(generated_content),
                        "pricing": pricing,
                        "model_used": model,
                        "timestamp": datetime.now().isoformat()
                    }

                    return formatted_result
                else:
                    return {"error": f"Unexpected response format: {result}"}

            else:
                return {"error": f"API request failed: {response.status_code} - {response.text}"}

        except requests.exceptions.RequestException as e:
            return {"error": f"Network error: {str(e)}"}
        except Exception as e:
            return {"error": f"Processing error: {str(e)}"}

    def get_session_stats(self) -> Dict[str, Any]:
        """Get current session statistics"""
        return {
            **self.session_stats,
            "uptime_minutes": round(
                (datetime.now() - datetime.fromisoformat(self.session_stats["start_time"])).total_seconds() / 60, 
                2
            )
        }

# Initialize the OASIS content generator
generator = OasisContentGenerator()

def generate_content_interface(content_type: str, topic: str, audience: str, tone: str, model: str) -> tuple:
    """Gradio interface function for content generation"""
    result = generator.generate_content(content_type, topic, audience, tone, model)

    if "error" in result:
        return result["error"], "", ""

    # Format content output
    content_output = f"""
    ## ðŸ“ Generated Content

    **Type:** {result['content_type'].replace('_', ' ').title()}
    **Topic:** {result['topic']}
    **Audience:** {result['audience']}
    **Tone:** {result['tone']}

    ---

    {result['generated_content']}

    ---

    **Statistics:**
    - Word Count: {result['word_count']}
    - Character Count: {result['character_count']}
    - Generated: {result['timestamp'][:19]}
    """

    # Format pricing output
    pricing_output = f"""
    ## ðŸ’° OASIS Content Pricing

    **{result['content_type'].replace('_', ' ').title()} Pricing:**

    ### Service Tiers:
    - **Starter:** ${result['pricing']['starter']:.2f}
    - **Professional:** ${result['pricing']['professional']:.2f}
    - **Enterprise:** ${result['pricing']['enterprise']:.2f}

    **What you get:**
    - Starter: Basic content generation
    - Professional: Enhanced quality + revisions
    - Enterprise: Custom models + priority support + bulk discounts

    *Ready to scale your content creation?*
    """

    # Format session stats
    stats = generator.get_session_stats()
    stats_output = f"""
    ## ðŸ“Š Session Performance

    - **Content Pieces:** {stats['content_generated']}
    - **Total Words:** {stats['total_tokens']}
    - **Revenue Generated:** ${stats['revenue_generated']:.2f}
    - **Session Uptime:** {stats['uptime_minutes']} minutes
    - **Avg. Revenue/Piece:** ${stats['revenue_generated']/max(1, stats['content_generated']):.2f}
    """

    return content_output, pricing_output, stats_output
