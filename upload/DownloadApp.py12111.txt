@dataclass
class ServiceInfo:
    """Service registration information"""
    service_id: str
    service_name: str
    service_url: str
    service_type: str  # 'api-processor', 'ui-interface', 'termux-client'
    status: str = "active"  # 'active', 'inactive', 'error'
    last_heartbeat: datetime = field(default_factory=datetime.now)
    response_time: float = 0.0
    total_requests: int = 0
    error_count: int = 0
    metadata: Dict = field(default_factory=dict)

@dataclass 
class TermuxClient:
    """Termux client information"""
    client_id: str
    session_id: str
    registration_time: datetime
    last_activity: datetime
    requests_made: int = 0
    revenue_generated: float = 0.0
    device_info: Dict = field(default_factory=dict)
    status: str = "active"

class OASISOrchestrator:
    """Main orchestrator for OASIS ecosystem"""

    def __init__(self):
        # Service registry
        self.services: Dict[str, ServiceInfo] = {}
        self.termux_clients: Dict[str, TermuxClient] = {}

        # System statistics
        self.system_stats = {
            "total_requests": 0,
            "total_revenue": 0.0,
            "active_services": 0,
            "active_clients": 0,
            "start_time": datetime.now(),
            "last_health_check": None,
            "error_count": 0,
            "avg_response_time": 0.0
        }

        # Request history for analytics
        self.request_history = deque(maxlen=1000)
        self.revenue_history = deque(maxlen=1000)

        # Health monitoring
        self.health_check_interval = 30  # seconds
        self.max_error_threshold = 10

        # Start background tasks
        self.start_background_monitoring()

        logger.info("OASIS Orchestrator initialized successfully")

    def start_background_monitoring(self):
        """Start background health monitoring"""
        def health_monitor():
            while True:
                try:
                    asyncio.run(self.check_service_health())
                    self.cleanup_inactive_clients()
                    time.sleep(self.health_check_interval)
                except Exception as e:
                    logger.error(f"Health monitor error: {str(e)}")
                    time.sleep(10)

        thread = threading.Thread(target=health_monitor, daemon=True)
        thread.start()
        logger.info("Background health monitoring started")

    async def check_service_health(self):
        """Check health of all registered services"""
        self.system_stats["last_health_check"] = datetime.now()
        active_count = 0
        total_response_time = 0

        for service_id, service in self.services.items():
            try:
                start_time = time.time()

                async with aiohttp.ClientSession(timeout=aiohttp.ClientTimeout(total=10)) as session:
                    async with session.get(f"{service.service_url}/health", timeout=5) as response:
                        if response.status == 200:
                            service.status = "active"
                            service.last_heartbeat = datetime.now()
                            active_count += 1
                        else:
                            service.status = "error"
                            service.error_count += 1

                response_time = time.time() - start_time
                service.response_time = response_time
                total_response_time += response_time

            except Exception as e:
                service.status = "error" 
                service.error_count += 1
                logger.warning(f"Service {service_id} health check failed: {str(e)}")

        self.system_stats["active_services"] = active_count
        if len(self.services) > 0:
            self.system_stats["avg_response_time"] = total_response_time / len(self.services)

    def cleanup_inactive_clients(self):
        """Remove inactive Termux clients"""
        cutoff_time = datetime.now() - timedelta(hours=2)
        inactive_clients = [
            client_id for client_id, client in self.termux_clients.items()
            if client.last_activity < cutoff_time
        ]

        for client_id in inactive_clients:
            del self.termux_clients[client_id]
            logger.info(f"Removed inactive client: {client_id}")

        self.system_stats["active_clients"] = len(self.termux_clients)

    def register_service(self, service_data: Dict[str, Any]) -> str:
        """Register a new service in the ecosystem"""
        service_id = str(uuid.uuid4())

        service = ServiceInfo(
            service_id=service_id,
            service_name=service_data.get("name", "unknown"),
            service_url=service_data.get("url", ""),
            service_type=service_data.get("type", "unknown"),
            metadata=service_data.get("metadata", {})
        )

        self.services[service_id] = service
        logger.info(f"Registered service: {service.service_name} ({service_id})")

        return service_id

    def register_termux_client(self, client_id: str, metadata: Dict = None) -> str:
        """Register a Termux client"""
        session_id = f"termux_{client_id}_{int(time.time())}"

        client = TermuxClient(
            client_id=client_id,
            session_id=session_id,
            registration_time=datetime.now(),
            last_activity=datetime.now(),
            device_info=metadata or {}
        )

        self.termux_clients[session_id] = client
        logger.info(f"Registered Termux client: {client_id} -> {session_id}")

        return session_id

    def proxy_request(self, service_type: str, endpoint: str, method: str = "GET", 
                     data: Dict = None) -> Dict[str, Any]:
        """Proxy requests to appropriate services with load balancing"""

        # Find available services of the requested type
        available_services = [
            service for service in self.services.values()
            if service.service_type == service_type and service.status == "active"
        ]

        if not available_services:
            raise HTTPException(
                status_code=503, 
                detail=f"No active {service_type} services available"
            )

        # Simple load balancing - choose service with lowest request count
        target_service = min(available_services, key=lambda s: s.total_requests)

        try:
            url = f"{target_service.service_url}{endpoint}"

            # Make the request
            if method == "POST":
                response = requests.post(url, json=data, timeout=30)
            else:
                response = requests.get(url, timeout=30)

            # Update service stats
            target_service.total_requests += 1
            target_service.last_heartbeat = datetime.now()

            # Update system stats
            self.system_stats["total_requests"] += 1

            # Track revenue if present in response
            if response.status_code == 200:
                result = response.json()
                if isinstance(result, dict) and "revenue_generated" in result:
                    revenue = result["revenue_generated"]
                    self.system_stats["total_revenue"] += revenue
                    self.revenue_history.append({
                        "timestamp": datetime.now().isoformat(),
                        "amount": revenue,
                        "service": target_service.service_name
                    })

                return {"success": True, "data": result, "service_used": target_service.service_name}
            else:
                target_service.error_count += 1
                self.system_stats["error_count"] += 1
                return {"success": False, "error": f"Service error: {response.status_code}"}

        except Exception as e:
            target_service.error_count += 1
            self.system_stats["error_count"] += 1
            logger.error(f"Request proxy error: {str(e)}")
            return {"success": False, "error": str(e)}

    def update_client_activity(self, session_id: str, revenue_delta: float = 0.0):
        """Update Termux client activity and revenue"""
        if session_id in self.termux_clients:
            client = self.termux_clients[session_id]
            client.last_activity = datetime.now()
            client.requests_made += 1
            client.revenue_generated += revenue_delta

    def get_system_overview(self) -> Dict[str, Any]:
        """Get comprehensive system overview"""
        uptime = datetime.now() - self.system_stats["start_time"]

        # Service status summary
        service_summary = defaultdict(int)
        for service in self.services.values():
            service_summary[service.status] += 1

        # Recent revenue trend
        recent_revenue = sum(
            entry["amount"] for entry in list(self.revenue_history)[-10:]
        ) if self.revenue_history else 0.0

        return {
            "system_info": {
                "uptime": str(uptime),
                "total_requests": self.system_stats["total_requests"],
                "total_revenue": round(self.system_stats["total_revenue"], 2),
                "recent_revenue": round(recent_revenue, 2),
                "error_rate": round(self.system_stats["error_count"] / max(self.system_stats["total_requests"], 1) * 100, 2)
            },
            "services": {
                "total": len(self.services),
                "active": service_summary.get("active", 0),
                "inactive": service_summary.get("inactive", 0),
                "error": service_summary.get("error", 0),
                "avg_response_time": round(self.system_stats["avg_response_time"], 3)
            },
            "clients": {
                "total_registered": len(self.termux_clients),
                "active": len([c for c in self.termux_clients.values() if c.status == "active"])
            },
            "performance": {
                "last_health_check": self.system_stats["last_health_check"].isoformat() if self.system_stats["last_health_check"] else None,
                "requests_per_minute": len([r for r in self.request_history if datetime.fromisoformat(r["timestamp"]) > datetime.now() - timedelta(minutes=1)]) if self.request_history else 0
            }
        }

                "content": "/proxy/content",

        raise HTTPException(status_code=400, detail=str(e))
