#!/bin/bash

# ==============================================================================
# OASIS 2.0 ULTRA-LIGHTWEIGHT TERMUX SETUP SCRIPT
# Zero Heavy Dependencies • Pure API-First • <5MB Footprint
# ==============================================================================

echo "🚀 OASIS 2.0 ULTRA-LIGHTWEIGHT SETUP STARTING..."
echo "================================================================="
echo "🎯 ZERO HEAVY DEPENDENCIES - PURE API APPROACH"
echo "📱 ULTRA-LIGHTWEIGHT - <5MB FOOTPRINT"
echo "☁️ HUGGING FACE POWERED - 100,000+ MODELS"
echo "💰 REVENUE READY - $1K→$10K/MONTH TARGET"
echo "================================================================="

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
PURPLE='\033[0;35m'
CYAN='\033[0;36m'
NC='\033[0m' # No Color

print_status() {
    echo -e "${GREEN}✅ $1${NC}"
}

print_warning() {
    echo -e "${YELLOW}⚠️ $1${NC}"
}

print_error() {
    echo -e "${RED}❌ $1${NC}"
}

print_info() {
    echo -e "${BLUE}ℹ️ $1${NC}"
}

print_header() {
    echo -e "${PURPLE}🔧 $1${NC}"
}

# Function to check if command exists
command_exists() {
    command -v "$1" >/dev/null 2>&1
}

# Phase 1: System Update and Core Package Installation
print_header "Phase 1: Core System Setup (Ultra-Minimal)"

# Update Termux packages
print_info "Updating Termux packages..."
pkg update -y && pkg upgrade -y
if [ $? -eq 0 ]; then
    print_status "Termux packages updated successfully"
else
    print_error "Failed to update Termux packages"
    exit 1
fi

# Install ONLY essential packages (NO HEAVY DEPENDENCIES!)
print_info "Installing essential packages ONLY..."
print_warning "EXPLICITLY AVOIDING: numpy, torch, pytorch, pandas, scikit-learn, transformers, gradio"

# Core essentials (<2MB total)
pkg install -y python curl git nano wget nodejs
if [ $? -eq 0 ]; then
    print_status "Essential packages installed (python, curl, git, nano, wget, nodejs)"
else
    print_error "Failed to install essential packages"
    exit 1
fi

# Verify Python installation
if command_exists python; then
    PYTHON_VERSION=$(python --version 2>&1)
    print_status "Python installed: $PYTHON_VERSION"
else
    print_error "Python installation failed"
    exit 1
fi

# Phase 2: Ultra-Lightweight Python Dependencies
print_header "Phase 2: Python Dependencies Setup (Ultra-Minimal)"

# Install ONLY lightweight HTTP client
print_info "Installing ultra-minimal Python packages..."
pip install --upgrade pip
pip install requests python-dotenv
if [ $? -eq 0 ]; then
    print_status "Lightweight packages installed (requests, python-dotenv)"
else
    print_error "Failed to install Python packages"
    exit 1
fi

# Verify package sizes
print_info "Verifying ultra-lightweight footprint..."
python -c "
import sys
import os
total_size = 0
for root, dirs, files in os.walk(sys.executable.split('/bin/python')[0] + '/lib/python3.11/site-packages'):
    for file in files:
        if any(pkg in root for pkg in ['requests', 'dotenv']):
            total_size += os.path.getsize(os.path.join(root, file))
print(f'Total Python packages size: {total_size/1024/1024:.2f} MB')
if total_size/1024/1024 < 3:
    print('✅ Ultra-lightweight requirement met (<3MB)')
else:
    print('⚠️ Package size exceeds ultra-lightweight threshold')
"

# Phase 3: OASIS Directory Structure
print_header "Phase 3: OASIS Directory Structure"

# Create OASIS directory structure
OASIS_HOME="$HOME/oasis_2.0"
mkdir -p "$OASIS_HOME"/{scripts,config,logs,data,clients,business}

print_status "OASIS directory structure created at: $OASIS_HOME"
print_info "Directory structure:"
echo "    📁 $OASIS_HOME/"
echo "    ├── 📁 scripts/     # Core Python scripts"
echo "    ├── 📁 config/      # Configuration files"
echo "    ├── 📁 logs/        # System logs"
echo "    ├── 📁 data/        # Data storage"
echo "    ├── 📁 clients/     # Client management"
echo "    └── 📁 business/    # Revenue tracking"

# Phase 4: Environment Configuration
print_header "Phase 4: Environment Configuration"

# Create .env template
cat > "$OASIS_HOME/config/.env" << EOF
# OASIS 2.0 Ultra-Lightweight Configuration
# Replace with your actual Hugging Face API key
HUGGINGFACE_API_KEY=hf_YOUR_API_KEY_HERE

# System Configuration
OASIS_VERSION=2.0.0
OASIS_ENVIRONMENT=termux
OASIS_LOG_LEVEL=INFO

# API Configuration
HF_API_BASE_URL=https://api-inference.huggingface.co/models
HF_TIMEOUT=60
HF_RETRY_ATTEMPTS=3

# Business Configuration
REVENUE_TARGET_MONTHLY=10000
PRICING_PER_REQUEST=0.01
PRICING_CONTENT_ARTICLE=150

# Security
JWT_SECRET=your_jwt_secret_key_here
API_RATE_LIMIT=1000

# Monitoring
METRICS_ENABLED=true
PERFORMANCE_TRACKING=true
BUSINESS_METRICS=true
EOF

print_status "Environment configuration created at: $OASIS_HOME/config/.env"
print_warning "IMPORTANT: Edit .env file and add your Hugging Face API key!"

# Phase 5: Installation Verification
print_header "Phase 5: Installation Verification"

# Check total footprint
TOTAL_SIZE=$(du -sm "$OASIS_HOME" | cut -f1)
print_info "OASIS installation size: ${TOTAL_SIZE}MB"

if [ "$TOTAL_SIZE" -lt 5 ]; then
    print_status "Ultra-lightweight requirement met (<5MB)"
else
    print_warning "Installation size exceeds 5MB threshold"
fi

# Verify essential commands
print_info "Verifying essential commands..."
for cmd in python curl git nano wget; do
    if command_exists "$cmd"; then
        print_status "$cmd: Available"
    else
        print_error "$cmd: Missing"
    fi
done

# Python packages verification
python -c "
try:
    import requests
    print('✅ requests: Available')
except ImportError:
    print('❌ requests: Missing')

try:
    from dotenv import load_dotenv
    print('✅ python-dotenv: Available')
except ImportError:
    print('❌ python-dotenv: Missing')

import json
import os
import sys
print('✅ Built-in modules: json, os, sys available')
print('✅ Zero heavy dependencies confirmed!')
"

# Phase 6: Quick Start Instructions
print_header "Phase 6: Next Steps"

echo ""
echo "🎉 OASIS 2.0 Ultra-Lightweight Setup Complete!"
echo ""
echo "📋 Next steps:"
echo "1. Edit config: nano $OASIS_HOME/config/.env"
echo "2. Add your Hugging Face API key"
echo "3. Copy OASIS scripts to: $OASIS_HOME/scripts/"
echo "4. Run: cd $OASIS_HOME && python scripts/oasis_core.py"
echo ""
echo "🚀 Ready for revolutionary AI automation!"
echo "💰 Target: $1K → $5K → $10K/month in 90 days"
echo ""
print_status "OASIS 2.0 Ultra-Lightweight setup completed successfully!"

# Create quick launcher script
cat > "$HOME/launch_oasis.sh" << 'EOF'
#!/bin/bash
cd ~/oasis_2.0
echo "🚀 Launching OASIS 2.0..."
python scripts/oasis_core.py
EOF
chmod +x "$HOME/launch_oasis.sh"

print_status "Quick launcher created: ~/launch_oasis.sh"
echo ""
echo "To start OASIS: ./launch_oasis.sh"
