class TermuxConfig:
    """Configuration management for Termux environment"""

    def __init__(self):
        self.config_file = "~/.oasis/config.json"
        self.config = self.load_config()

    def load_config(self) -> Dict[str, Any]:
        """Load configuration from file"""
        config_path = os.path.expanduser(self.config_file)

        if os.path.exists(config_path):
            try:
                with open(config_path, 'r') as f:
                    return json.load(f)
            except Exception:
                pass

        # Default configuration
        return {
            "spaces": {
                "sentiment": "elmatador0197-my-oasis-agent",
                "content_gen": "elmatador0197-content-factory",
                "dashboard": "elmatador0197-oasis-dashboard"
            },
            "api": {
                "gateway_url": "https://api.oasis-ai.com",
                "hf_token": os.getenv("HF_TOKEN", ""),
                "rate_limits": {
                    "starter": 100,
                    "professional": 1000,
                    "enterprise": 10000
                }
            },
            "mobile": {
                "notifications_enabled": True,
                "auto_sync": True,
                "data_saver_mode": False,
                "offline_mode": False
            },
            "deployment": {
                "github_repo": "elmatador0197/oasis-v3",
                "auto_deploy": True,
                "branch": "main",
                "deploy_on_push": True
            }
        }

    def save_config(self) -> None:
        """Save configuration to file"""
        config_path = os.path.expanduser(self.config_file)
        os.makedirs(os.path.dirname(config_path), exist_ok=True)

        with open(config_path, 'w') as f:
            json.dump(self.config, f, indent=2)

    def get(self, key: str, default: Any = None) -> Any:
        """Get configuration value using dot notation"""
        keys = key.split('.')
        value = self.config

        for k in keys:
            if isinstance(value, dict) and k in value:
                value = value[k]
            else:
                return default

        return value

    def set(self, key: str, value: Any) -> None:
        """Set configuration value using dot notation"""
        keys = key.split('.')
        config = self.config

        for k in keys[:-1]:
            if k not in config:
                config[k] = {}
            config = config[k]

        config[keys[-1]] = value
        self.save_config()

    def interactive_shell(self) -> None:
        """Start interactive command shell"""
        print("ðŸŒŸ OASIS v3 Termux Command Center")
        print("Type 'help' for available commands, 'exit' to quit")
        print()

        commands = {
            'status': self.status,
            'deploy': lambda space: self.deploy_space(space) if space else {"error": "Specify space name"},
            'monitor': self.monitor_resources,
            'sync': self.sync_data,
            'report': lambda: print(self.generate_report()),
            'config': lambda key, val=None: self.config.set(key, val) if val else self.config.get(key),
            'help': lambda: print("""
Available Commands:
â€¢ status          - Show system status
â€¢ deploy <space>  - Deploy Hugging Face Space
â€¢ monitor        - Show resource monitoring
â€¢ sync           - Synchronize data
â€¢ report         - Generate status report
â€¢ config <key> [val] - Get/set configuration
â€¢ exit           - Exit command center
            """)
        }

        while True:
            try:
                cmd_line = input("oasis> ").strip()
                if not cmd_line:
                    continue

                if cmd_line == 'exit':
                    print("ðŸ‘‹ Goodbye!")
                    break

                parts = cmd_line.split()
                cmd = parts[0]
                args = parts[1:]

                if cmd in commands:
                    try:
                        result = commands[cmd](*args)
                        if result and cmd != 'report':
                            print(json.dumps(result, indent=2))
                    except Exception as e:
                        print(f"âŒ Error: {e}")
                else:
                    print(f"âŒ Unknown command: {cmd}. Type 'help' for available commands.")

            except KeyboardInterrupt:
                print("\nðŸ‘‹ Goodbye!")
                break
            except EOFError:
                break

def main():
    """Main entry point for Termux manager"""
    parser = argparse.ArgumentParser(description="OASIS v3 Termux Command Center")
    parser.add_argument('--version', action='version', version='OASIS v3.0')

    subparsers = parser.add_subparsers(dest='command', help='Available commands')

    # Status command
    subparsers.add_parser('status', help='Show system status')

    # Deploy command
    deploy_parser = subparsers.add_parser('deploy', help='Deploy Hugging Face Space')
    deploy_parser.add_argument('space', help='Space name to deploy')

    # Monitor command
    subparsers.add_parser('monitor', help='Monitor resources')

    # Sync command
    subparsers.add_parser('sync', help='Synchronize data')

    # Report command
    subparsers.add_parser('report', help='Generate status report')

    # Interactive shell
    subparsers.add_parser('shell', help='Start interactive shell')

    args = parser.parse_args()

    manager = OasisTermuxManager()

    if args.command == 'status':
        print(json.dumps(manager.status(), indent=2))
    elif args.command == 'deploy':
        result = manager.deploy_space(args.space)
        print(json.dumps(result, indent=2))
    elif args.command == 'monitor':
        print(json.dumps(manager.monitor_resources(), indent=2))
    elif args.command == 'sync':
        result = manager.sync_data()
        print(json.dumps(result, indent=2))
    elif args.command == 'report':
        print(manager.generate_report())
    elif args.command == 'shell':
        manager.interactive_shell()
    else:
        # No command specified, show help and start interactive shell
        parser.print_help()
        print()
        manager.interactive_shell()

if __name__ == "__main__":
    main()"
