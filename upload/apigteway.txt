class RateLimiter:
    """Token bucket rate limiter"""

    def __init__(self, requests_per_minute=60):
        self.capacity = requests_per_minute
        self.tokens = requests_per_minute
        self.refill_rate = requests_per_minute / 60.0  # tokens per second
        self.last_refill = time.time()
        self.client_buckets = defaultdict(lambda: {
            'tokens': requests_per_minute,
            'last_refill': time.time()
        })

    def allow_request(self, client_id):
        """Check if request is allowed for client"""
        now = time.time()
        bucket = self.client_buckets[client_id]

        # Refill tokens
        time_passed = now - bucket['last_refill']
        tokens_to_add = time_passed * self.refill_rate
        bucket['tokens'] = min(self.capacity, bucket['tokens'] + tokens_to_add)
        bucket['last_refill'] = now

        # Check if request allowed
        if bucket['tokens'] >= 1:
            bucket['tokens'] -= 1
            return True
        return False

class AuthManager:
    """Simple API key authentication"""

    def __init__(self, keys_file='config/api_keys.json'):
        self.keys_file = keys_file
        self.api_keys = self.load_api_keys()
        self.rate_limiters = {}

    def load_api_keys(self):
        """Load API keys from file"""
        try:
            if os.path.exists(self.keys_file):
                with open(self.keys_file, 'r') as f:
                    return json.load(f)
        except:
            pass

        # Create default API keys
        default_keys = {
            'admin_key_123': {
                'name': 'Admin Key',
                'permissions': ['*'],
                'rate_limit': 1000,
                'created': datetime.now().isoformat()
            },
            'demo_key_456': {
                'name': 'Demo Key', 
                'permissions': ['read'],
                'rate_limit': 100,
                'created': datetime.now().isoformat()
            }
        }

        self.save_api_keys(default_keys)
        return default_keys

    def save_api_keys(self, keys):
        """Save API keys to file"""
        try:
            os.makedirs(os.path.dirname(self.keys_file), exist_ok=True)
            with open(self.keys_file, 'w') as f:
                json.dump(keys, f, indent=2)
        except Exception as e:
            print(f"Error saving API keys: {e}")

    def authenticate(self, api_key):
        """Authenticate API key"""
        if api_key in self.api_keys:
            key_info = self.api_keys[api_key]

            # Initialize rate limiter for this key
            if api_key not in self.rate_limiters:
                self.rate_limiters[api_key] = RateLimiter(key_info.get('rate_limit', 60))

            return key_info
        return None

    def check_permission(self, key_info, action):
        """Check if API key has permission for action"""
        permissions = key_info.get('permissions', [])
        return '*' in permissions or action in permissions

    def rate_limit_check(self, api_key):
        """Check rate limit for API key"""
        if api_key in self.rate_limiters:
            return self.rate_limiters[api_key].allow_request(api_key)
        return False

class MetricsCollector:
    """Collect API usage metrics"""

    def __init__(self):
        self.request_history = deque(maxlen=10000)
        self.endpoint_stats = defaultdict(lambda: {
            'count': 0,
            'total_time': 0,
            'errors': 0,
            'last_request': None
        })

    def record_request(self, endpoint, method, status_code, response_time, api_key=None):
        """Record API request metrics"""
        timestamp = time.time()

        request_data = {
            'timestamp': timestamp,
            'endpoint': endpoint,
            'method': method,
            'status_code': status_code,
            'response_time': response_time,
            'api_key': api_key[:8] + '...' if api_key else 'none'  # Partially mask key
        }

        self.request_history.append(request_data)

        # Update endpoint stats
        stats = self.endpoint_stats[endpoint]
        stats['count'] += 1
        stats['total_time'] += response_time
        stats['last_request'] = timestamp

        if status_code >= 400:
            stats['errors'] += 1

    def get_metrics_summary(self):
        """Get metrics summary"""
        now = time.time()

        # Recent requests (last hour)
        hour_ago = now - 3600
        recent_requests = [r for r in self.request_history if r['timestamp'] > hour_ago]

        # Calculate stats
        total_requests = len(recent_requests)
        error_requests = len([r for r in recent_requests if r['status_code'] >= 400])
        error_rate = (error_requests / max(total_requests, 1)) * 100

        avg_response_time = 0
        if recent_requests:
            avg_response_time = sum(r['response_time'] for r in recent_requests) / len(recent_requests)

        return {
            'total_requests_hour': total_requests,
            'error_rate_percent': error_rate,
            'average_response_time': avg_response_time,
            'endpoint_stats': dict(self.endpoint_stats),
            'timestamp': datetime.now().isoformat()
        }

class APIGatewayHandler(BaseHTTPRequestHandler):
    """HTTP request handler for API Gateway"""

    def __init__(self, *args, auth_manager=None, service_registry=None, metrics_collector=None, **kwargs):
        self.auth_manager = auth_manager
        self.service_registry = service_registry
        self.metrics_collector = metrics_collector
        super().__init__(*args, **kwargs)

    def log_message(self, format, *args):
        """Override to suppress default logging"""
        pass

    def do_GET(self):
        """Handle GET requests"""
        start_time = time.time()

        try:
            # Parse URL and query parameters
            parsed = urlparse(self.path)
            path = parsed.path
            params = parse_qs(parsed.query)

            # Authentication
            api_key = self.headers.get('X-API-Key') or params.get('api_key', [None])[0]
            auth_info = self._authenticate(api_key)

            if not auth_info and path not in ['/api/docs', '/api/health']:
                self._send_error(401, 'Authentication required')
                return

            # Route requests
            if path == '/api/docs':
                self._handle_docs()
            elif path == '/api/health':
                self._handle_health()
            elif path == '/api/v1/status':
                self._handle_status()
            elif path == '/api/v1/metrics':
                self._handle_metrics()
            elif path == '/api/v1/clients':
                self._handle_clients()
            elif path == '/api/v1/revenue':
                self._handle_revenue()
            else:
                self._send_error(404, 'Endpoint not found')

        except Exception as e:
            self._send_error(500, f'Internal server error: {str(e)}')
        finally:
            # Record metrics
            response_time = time.time() - start_time
            status_code = getattr(self, '_response_status', 500)
            self.metrics_collector.record_request(
                self.path, 'GET', status_code, response_time, api_key
            )

    def do_POST(self):
        """Handle POST requests"""
        start_time = time.time()

        try:
            # Parse URL
            parsed = urlparse(self.path)
            path = parsed.path

            # Authentication
            api_key = self.headers.get('X-API-Key')
            auth_info = self._authenticate(api_key)

            if not auth_info:
                self._send_error(401, 'Authentication required')
                return

            # Read request body
            content_length = int(self.headers.get('Content-Length', 0))
            if content_length > 0:
                body = self.rfile.read(content_length).decode('utf-8')
                try:
                    request_data = json.loads(body)
                except:
                    request_data = {}
            else:
                request_data = {}

            # Route requests
            if path == '/api/v1/generate':
                self._handle_generate(request_data)
            elif path == '/api/v1/sentiment':
                self._handle_sentiment(request_data)
            elif path == '/api/v1/translate':
                self._handle_translate(request_data)
            elif path == '/api/v1/services':
                self._handle_service_request(request_data)
            else:
                self._send_error(404, 'Endpoint not found')

        except Exception as e:
            self._send_error(500, f'Internal server error: {str(e)}')
        finally:
            # Record metrics
            response_time = time.time() - start_time
            status_code = getattr(self, '_response_status', 500)
            self.metrics_collector.record_request(
                self.path, 'POST', status_code, response_time, api_key
            )

    def _authenticate(self, api_key):
        """Authenticate request"""
        if not api_key:
            return None

        auth_info = self.auth_manager.authenticate(api_key)
        if not auth_info:
            return None

        # Check rate limit
        if not self.auth_manager.rate_limit_check(api_key):
            return None

        return auth_info

    def _send_json_response(self, data, status_code=200):
        """Send JSON response"""
        self._response_status = status_code
        response = json.dumps(data, indent=2)

        self.send_response(status_code)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Content-Length', len(response))
        self.send_header('Access-Control-Allow-Origin', '*')
        self.end_headers()

        self.wfile.write(response.encode('utf-8'))

    def _send_error(self, status_code, message):
        """Send error response"""
        self._response_status = status_code
        error_response = {
            'error': message,
            'status_code': status_code,
            'timestamp': datetime.now().isoformat()
        }
        self._send_json_response(error_response, status_code)

class APIGateway:
    """Main API Gateway server"""

    def __init__(self, host='0.0.0.0', port=8000):
        self.host = host
        self.port = port
        self.auth_manager = AuthManager()
        self.service_registry = ServiceRegistry()
        self.metrics_collector = MetricsCollector()
        self.server = None
        self.running = False

    def create_handler(self):
        """Create request handler with dependencies"""
        def handler(*args, **kwargs):
            return APIGatewayHandler(
                *args, 
                auth_manager=self.auth_manager,
                service_registry=self.service_registry,
                metrics_collector=self.metrics_collector,
                **kwargs
            )
        return handler

    def start(self):
        """Start the API Gateway server"""
        try:
            handler_class = self.create_handler()
            self.server = HTTPServer((self.host, self.port), handler_class)
            self.server.start_time = time.time()

            self.running = True

            print(f"ðŸš€ OASIS API Gateway started")
            print(f"   Server: http://{self.host}:{self.port}")
            print(f"   Documentation: http://{self.host}:{self.port}/api/docs")
            print(f"   Health Check: http://{self.host}:{self.port}/api/health")
            print()
            print("ðŸ“š API Keys:")
            for key, info in self.auth_manager.api_keys.items():
                print(f"   {info['name']}: {key}")
            print()
            print("ðŸ›‘ Press Ctrl+C to stop")

            self.server.serve_forever()

        except KeyboardInterrupt:
            self.stop()
        except Exception as e:
            print(f"âŒ Server error: {e}")
            self.running = False

    def stop(self):
        """Stop the API Gateway server"""
        if self.server:
            print("\nðŸ›‘ Stopping API Gateway...")
            self.server.shutdown()
            self.server.server_close()
            self.running = False
            print("âœ… API Gateway stopped")
