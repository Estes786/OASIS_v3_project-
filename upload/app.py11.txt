    async def process_sentiment(self, text: str, model: str = "sentiment") -> Dict[str, Any]:
        """Advanced sentiment analysis with confidence scores"""
        start_time = time.time()

        try:
            # Use local pipeline or fallback to HF API
            if self.sentiment_pipeline:
                results = self.sentiment_pipeline(text)

                # Process results
                sentiment_data = {
                    "predictions": results[0] if results else [],
                    "dominant_sentiment": results[0][0]["label"] if results else "NEUTRAL",
                    "confidence": results[0][0]["score"] if results else 0.5,
                    "analysis_method": "local_pipeline"
                }
            else:
                # Fallback to HF API
                sentiment_data = await self._api_fallback("sentiment-analysis", text)

            # Track usage and revenue
            processing_time = time.time() - start_time
            self._track_usage("sentiment", processing_time)

            return {
                "sentiment": sentiment_data,
                "processing_time": processing_time,
                "model_used": model,
                "text_length": len(text)
            }

        except Exception as e:
            logger.error(f"Sentiment analysis error: {e}")
            return {
                "error": str(e),
                "processing_time": time.time() - start_time
            }

    async def process_image(self, image_data: bytes) -> Dict[str, Any]:
        """Comprehensive image processing and analysis"""
        start_time = time.time()

        try:
            # Convert bytes to PIL Image
            image = Image.open(io.BytesIO(image_data))

            classification_results = self.image_classifier(image)

            # Image captioning
            caption_results = self.image_captioning(image)

            # Image metadata
            metadata = {
                "size": image.size,
                "mode": image.mode,
                "format": getattr(image, 'format', 'Unknown')
            }

            processing_time = time.time() - start_time
            self._track_usage("image_processing", processing_time)

            return {
                "classification": classification_results[:5],  # Top 5 predictions
                "caption": caption_results[0]["generated_text"] if caption_results else "No caption generated",
                "metadata": metadata,
                "processing_time": processing_time
            }

        except Exception as e:
            logger.error(f"Image processing error: {e}")
            return {
                "error": str(e),
                "processing_time": time.time() - start_time
            }

    async def process_batch(self, requests: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        """Batch processing for multiple AI tasks"""
        start_time = time.time()
        results = []

        for req in requests:
            task_type = req.get("type", "sentiment")
            data = req.get("data", "")

            if task_type == "sentiment":
                result = await self.process_sentiment(data)
            elif task_type == "image":
                result = await self.process_image(data)
            else:
                result = {"error": f"Unknown task type: {task_type}"}

            results.append({
                "request_id": req.get("id", str(uuid.uuid4())[:8]),
                "type": task_type,
                "result": result
            })

        batch_time = time.time() - start_time
        self._track_usage("batch_processing", batch_time, count=len(requests))

        return {
            "batch_results": results,
            "total_requests": len(requests),
            "batch_processing_time": batch_time,
            "average_time_per_request": batch_time / len(requests) if requests else 0
        }

    def _track_usage(self, operation: str, processing_time: float, count: int = 1):
        """Track usage statistics and revenue"""
        self.request_count += count
        self.total_processing_time += processing_time

        # Revenue calculation (example pricing)
        pricing = {
            "sentiment": 0.001,
            "image_processing": 0.005,
            "batch_processing": 0.01
        }

        revenue = pricing.get(operation, 0.001) * count
        self.revenue_data["total_revenue"] += revenue
        self.revenue_data["total_requests"] += count

        # Daily tracking
        today = datetime.now().strftime("%Y-%m-%d")
        if today not in self.revenue_data["daily_revenue"]:
            self.revenue_data["daily_revenue"][today] = 0.0
        self.revenue_data["daily_revenue"][today] += revenue

        # Model usage tracking
        if operation not in self.revenue_data["model_usage"]:
            self.revenue_data["model_usage"][operation] = {"count": 0, "revenue": 0.0}
        self.revenue_data["model_usage"][operation]["count"] += count
        self.revenue_data["model_usage"][operation]["revenue"] += revenue
