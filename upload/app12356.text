# ===== REVENUE TRACKING SYSTEM =====
class RevenueEngine:
    """Advanced Revenue Tracking and Optimization"""

    def __init__(self):
        self.daily_revenue = 0.0
        self.monthly_revenue = 0.0
        self.total_requests = 0
        self.revenue_history = deque(maxlen=30)  # 30 days
        self.request_log = deque(maxlen=10000)   # Last 10K requests
        self.subscription_tiers = {
            "free": {"limit": 100, "price": 0},
            "pro": {"limit": 5000, "price": 29.99},
            "enterprise": {"limit": 50000, "price": 299.99}
        }
        self.active_subscriptions = defaultdict(int)
        self.lock = threading.Lock()

    def record_transaction(self, service_type: str, amount: float, user_id: str = None):
        """Record revenue transaction"""
        with self.lock:
            self.daily_revenue += amount
            self.monthly_revenue += amount
            self.total_requests += 1

            transaction = {
                "timestamp": datetime.now().isoformat(),
                "service": service_type,
                "amount": amount,
                "user_id": user_id,
                "cumulative": self.monthly_revenue
            }
            self.request_log.append(transaction)

            logger.info(f"Revenue +${amount:.3f} | Service: {service_type} | Total: ${self.monthly_revenue:.2f}")
            return transaction

    def get_revenue_stats(self) -> Dict[str, Any]:
        """Get comprehensive revenue statistics"""
        with self.lock:
            avg_per_request = self.monthly_revenue / max(self.total_requests, 1)
            projected_monthly = (self.daily_revenue * 30) if self.daily_revenue > 0 else 0

            return {
                "current_month_revenue": round(self.monthly_revenue, 2),
                "daily_revenue": round(self.daily_revenue, 2),
                "total_requests": self.total_requests,
                "avg_revenue_per_request": round(avg_per_request, 3),
                "projected_monthly": round(projected_monthly, 2),
                "target_progress": round((self.monthly_revenue / config.MONTHLY_REVENUE_TARGET) * 100, 1),
                "active_subscriptions": dict(self.active_subscriptions),
                "last_24h_requests": len([r for r in self.request_log 
                                        if (datetime.now() - datetime.fromisoformat(r["timestamp"])).days == 0])
            }

    def optimize_pricing(self) -> Dict[str, float]:
        """AI-driven pricing optimization"""
        if len(self.request_log) < 100:
            return config.API_PRICING

        # Analyze request patterns
        recent_requests = list(self.request_log)[-100:]
        service_performance = defaultdict(list)

        for req in recent_requests:
            service_performance[req["service"]].append(req["amount"])

        optimized_pricing = {}
        for service, base_price in config.API_PRICING.items():
            if service in service_performance:
                avg_revenue = np.mean(service_performance[service])
                demand_factor = len(service_performance[service]) / 100
                # Dynamic pricing based on demand and performance
                optimized_pricing[service] = round(base_price * (1 + demand_factor * 0.1), 3)
            else:
                optimized_pricing[service] = base_price

        return optimized_pricing

# ===== PERFORMANCE MONITOR =====
class PerformanceMonitor:
    """Real-time Performance Monitoring"""

    def __init__(self):
        self.metrics = {
            "requests_per_minute": deque(maxlen=60),
            "response_times": deque(maxlen=1000),
            "error_count": 0,
            "success_count": 0,
            "cpu_usage": deque(maxlen=60),
            "memory_usage": deque(maxlen=60)
        }
        self.start_monitoring()

    def start_monitoring(self):
        """Start background monitoring"""
        def monitor():
            while True:
                try:
                    cpu_percent = psutil.cpu_percent()
                    memory_percent = psutil.virtual_memory().percent

                    self.metrics["cpu_usage"].append(cpu_percent)
                    self.metrics["memory_usage"].append(memory_percent)

                    time.sleep(1)
                except Exception as e:
                    logger.error(f"Monitoring error: {e}")

        thread = threading.Thread(target=monitor, daemon=True)
        thread.start()

    def record_request(self, response_time: float, success: bool = True):
        """Record request metrics"""
        self.metrics["response_times"].append(response_time)
        if success:
            self.metrics["success_count"] += 1
        else:
            self.metrics["error_count"] += 1

    def get_performance_stats(self) -> Dict[str, Any]:
        """Get current performance statistics"""
        return {
            "avg_response_time": np.mean(list(self.metrics["response_times"])) if self.metrics["response_times"] else 0,
            "requests_per_minute": len(self.metrics["requests_per_minute"]),
            "success_rate": (self.metrics["success_count"] / max(1, self.metrics["success_count"] + self.metrics["error_count"])) * 100,
            "cpu_usage": np.mean(list(self.metrics["cpu_usage"])) if self.metrics["cpu_usage"] else 0,
            "memory_usage": np.mean(list(self.metrics["memory_usage"])) if self.metrics["memory_usage"] else 0,
            "total_requests": self.metrics["success_count"] + self.metrics["error_count"]
        }

performance_monitor = PerformanceMonitor()
