"#!/usr/bin/env python3
"""
OASIS v3 Performance Benchmark
Tests orchestrator performance and mobile compatibility
"""

import sys
import json
import time
import urllib.request
import urllib.parse
from typing import Dict, List, Any

class OASISBenchmark:
    """Performance benchmarking for OASIS v3"""

    def __init__(self, config_file: str = "oasis_config.json"):
        """Initialize benchmark suite"""
        self.config_file = config_file
        self.results = []

    def benchmark_import_speed(self) -> Dict[str, float]:
        """Benchmark import performance"""
        print("ðŸƒ Benchmarking Import Speed...")

        # Test standard library imports
        std_imports = [
            'json', 'urllib.request', 'urllib.parse', 'urllib.error',
            'sys', 'os', 'time', 're', 'base64'
        ]

        import_times = {}

        for lib in std_imports:
            start_time = time.time()
            try:
                __import__(lib)
                import_time = (time.time() - start_time) * 1000  # Convert to ms
                import_times[lib] = import_time
                print(f"  ðŸ“¦ {lib}: {import_time:.2f}ms")
            except Exception as e:
                import_times[lib] = -1
                print(f"  âŒ {lib}: Failed ({e})")

        total_time = sum(t for t in import_times.values() if t > 0)
        print(f"  ðŸ† Total import time: {total_time:.2f}ms")

        return import_times

    def benchmark_orchestrator_init(self) -> Dict[str, float]:
        """Benchmark orchestrator initialization"""
        print("\nðŸƒ Benchmarking Orchestrator Initialization...")

        init_results = {}

        try:
            sys.path.insert(0, '.')
            from oasis_orchestrator import OASISOrchestrator

            # Multiple initialization runs
            runs = 5
            times = []

            for i in range(runs):
                start_time = time.time()
                orchestrator = OASISOrchestrator(self.config_file)
                init_time = (time.time() - start_time) * 1000
                times.append(init_time)
                print(f"  ðŸ”„ Run {i+1}: {init_time:.2f}ms")

            init_results['min_time'] = min(times)
            init_results['max_time'] = max(times)
            init_results['avg_time'] = sum(times) / len(times)
            init_results['runs'] = runs

            print(f"  ðŸ“Š Average: {init_results['avg_time']:.2f}ms")
            print(f"  ðŸ† Best: {init_results['min_time']:.2f}ms")
            print(f"  ðŸŒ Worst: {init_results['max_time']:.2f}ms")

        except Exception as e:
            print(f"  âŒ Initialization failed: {e}")
            init_results['error'] = str(e)

        return init_results

    def benchmark_memory_usage(self) -> Dict[str, Any]:
        """Estimate memory usage (basic approach)"""
        print("\nðŸ§  Benchmarking Memory Usage...")

        memory_results = {}

        try:
            import os

            # Get file sizes as proxy for memory usage
            files_to_check = [
                'oasis_orchestrator.py',
                'oasis_config.json'
            ]

            total_size = 0
            file_sizes = {}

            for filename in files_to_check:
                try:
                    size = os.path.getsize(filename)
                    file_sizes[filename] = size
                    total_size += size
                    print(f"  ðŸ“„ {filename}: {size:,} bytes ({size/1024:.1f}KB)")
                except:
                    file_sizes[filename] = 0

            memory_results['file_sizes'] = file_sizes
            memory_results['total_size'] = total_size
            memory_results['total_kb'] = total_size / 1024
            memory_results['total_mb'] = total_size / (1024 * 1024)

            print(f"  ðŸ’¾ Total footprint: {memory_results['total_kb']:.1f}KB ({memory_results['total_mb']:.2f}MB)")

            # Memory efficiency rating
            if memory_results['total_mb'] < 0.1:
                rating = "Excellent (Ultra-light)"
            elif memory_results['total_mb'] < 0.5:
                rating = "Good (Mobile-friendly)"
            elif memory_results['total_mb'] < 1.0:
                rating = "Fair (Acceptable)"
            else:
                rating = "Poor (Too heavy for mobile)"

            memory_results['efficiency_rating'] = rating
            print(f"  ðŸ… Efficiency: {rating}")

        except Exception as e:
            print(f"  âŒ Memory benchmark failed: {e}")
            memory_results['error'] = str(e)

        return memory_results

    def benchmark_network_latency(self) -> Dict[str, float]:
        """Benchmark network request latency"""
        print("\nðŸŒ Benchmarking Network Latency...")

        latency_results = {}

        test_endpoints = [
            ("DNS Test", "https://dns.google/resolve?name=google.com&type=A"),
            ("HF API", "https://api-inference.huggingface.co/"),
        ]

        for name, url in test_endpoints:
            try:
                # Multiple requests for average
                times = []
                for i in range(3):
                    start_time = time.time()
                    req = urllib.request.Request(url)
                    req.add_header('User-Agent', 'OASIS-Benchmark/1.0')

                    with urllib.request.urlopen(req, timeout=10) as response:
                        latency = (time.time() - start_time) * 1000
                        times.append(latency)

                avg_latency = sum(times) / len(times)
                latency_results[name] = avg_latency
                print(f"  ðŸš€ {name}: {avg_latency:.0f}ms (avg of {len(times)} requests)")

            except Exception as e:
                latency_results[name] = -1
                print(f"  âŒ {name}: Failed ({e})")

        return latency_results

    def benchmark_json_processing(self) -> Dict[str, float]:
        """Benchmark JSON processing performance"""
        print("\nðŸ“ Benchmarking JSON Processing...")

        json_results = {}

        try:
            # Test JSON operations with config file
            with open(self.config_file, 'r') as f:
                config_data = f.read()

            # Parse benchmark
            parse_times = []
            for i in range(100):
                start_time = time.time()
                data = json.loads(config_data)
                parse_time = (time.time() - start_time) * 1000000  # microseconds
                parse_times.append(parse_time)

            # Stringify benchmark
            stringify_times = []
            for i in range(100):
                start_time = time.time()
                json_str = json.dumps(data)
                stringify_time = (time.time() - start_time) * 1000000  # microseconds
                stringify_times.append(stringify_time)

            json_results['parse_avg_us'] = sum(parse_times) / len(parse_times)
            json_results['stringify_avg_us'] = sum(stringify_times) / len(stringify_times)
            json_results['config_size'] = len(config_data)

            print(f"  ðŸ“– JSON Parse: {json_results['parse_avg_us']:.1f}Î¼s avg")
            print(f"  ðŸ“ JSON Stringify: {json_results['stringify_avg_us']:.1f}Î¼s avg")
            print(f"  ðŸ“ Config size: {json_results['config_size']} bytes")

        except Exception as e:
            print(f"  âŒ JSON benchmark failed: {e}")
            json_results['error'] = str(e)

        return json_results

    def run_full_benchmark(self) -> Dict[str, Any]:
        """Run complete benchmark suite"""
        print("ðŸ OASIS v3 Performance Benchmark")
        print("=" * 50)

        start_time = time.time()

        # Run all benchmarks
        results = {
            'import_speed': self.benchmark_import_speed(),
            'orchestrator_init': self.benchmark_orchestrator_init(),
            'memory_usage': self.benchmark_memory_usage(),
            'network_latency': self.benchmark_network_latency(),
            'json_processing': self.benchmark_json_processing()
        }

        total_time = time.time() - start_time

        # Generate performance summary
        print("\n" + "=" * 50)
        print("ðŸ“Š PERFORMANCE SUMMARY")
        print("=" * 50)

        # Overall performance score
        score = 100
        issues = []

        # Check initialization time
        init_avg = results.get('orchestrator_init', {}).get('avg_time', 0)
        if init_avg > 1000:  # > 1 second
            score -= 20
            issues.append("Slow initialization")
        elif init_avg > 500:  # > 0.5 seconds
            score -= 10
            issues.append("Moderate initialization time")

        # Check memory usage
        memory_mb = results.get('memory_usage', {}).get('total_mb', 0)
        if memory_mb > 1.0:  # > 1MB
            score -= 30
            issues.append("High memory usage")
        elif memory_mb > 0.5:  # > 0.5MB
            score -= 15
            issues.append("Moderate memory usage")

        # Check network latency
        avg_latency = 0
        latency_count = 0
        for endpoint, latency in results.get('network_latency', {}).items():
            if latency > 0:
                avg_latency += latency
                latency_count += 1

        if latency_count > 0:
            avg_latency /= latency_count
            if avg_latency > 2000:  # > 2 seconds
                score -= 20
                issues.append("High network latency")

        # Performance rating
        if score >= 90:
            rating = "ðŸ† Excellent"
            color = "ðŸŸ¢"
        elif score >= 75:
            rating = "ðŸ‘ Good"
            color = "ðŸŸ¡"
        elif score >= 60:
            rating = "âš ï¸ Fair"
            color = "ðŸŸ "
        else:
            rating = "âŒ Poor"
            color = "ðŸ”´"

        print(f"{color} Overall Score: {score}/100 - {rating}")
        print(f"â±ï¸ Benchmark Duration: {total_time:.2f}s")

        if issues:
            print("\nâš ï¸ Performance Issues Detected:")
            for issue in issues:
                print(f"  â€¢ {issue}")
        else:
            print("\nâœ… No performance issues detected!")

        # Mobile compatibility check
        print("\nðŸ“± Mobile Compatibility:")
        if memory_mb < 0.5 and init_avg < 500:
            print("âœ… Excellent mobile compatibility")
        elif memory_mb < 1.0 and init_avg < 1000:
            print("âœ… Good mobile compatibility")
        else:
            print("âš ï¸ May have mobile compatibility issues")

        results['summary'] = {
            'score': score,
            'rating': rating,
            'issues': issues,
            'benchmark_time': total_time,
            'mobile_compatible': memory_mb < 1.0 and init_avg < 1000
        }

        return results

def main():
    """Main benchmark function"""
    import sys

    config_file = "oasis_config.json"
    if len(sys.argv) > 1:
        config_file = sys.argv[1]

    benchmark = OASISBenchmark(config_file)
    results = benchmark.run_full_benchmark()

    # Save results
    with open('benchmark_results.json', 'w') as f:
        json.dump(results, f, indent=2)

    print(f"\nðŸ’¾ Results saved to benchmark_results.json")

if __name__ == "__main__":
    main()"
