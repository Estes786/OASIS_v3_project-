name: 💰 Business Analytics & Revenue Monitoring

on:
  schedule:
    # Run analytics every 6 hours
    - cron: '0 */6 * * *'
    # Generate daily business reports at 8 AM UTC
    - cron: '0 8 * * *'
    # Weekly comprehensive analysis on Sundays at 6 AM UTC
    - cron: '0 6 * * 0'
  push:
    branches: [main]
    paths: ['scripts/analytics/**', 'database/analytics_views/**']
  workflow_dispatch:
    inputs:
      report_type:
        description: 'Type of analytics report'
        required: true
        default: 'daily'
        type: choice
        options: ['hourly', 'daily', 'weekly', 'monthly', 'custom']
      include_forecasting:
        description: 'Include revenue forecasting'
        required: false
        default: true
        type: boolean

env:
  SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
  SUPABASE_ANON_KEY: ${{ secrets.SUPABASE_ANON_KEY }}
  SUPABASE_SERVICE_ROLE_KEY: ${{ secrets.SUPABASE_SERVICE_ROLE_KEY }}
  ANALYTICS_WEBHOOK: ${{ secrets.ANALYTICS_WEBHOOK }}

jobs:
  collect_metrics:
    name: 📊 Collect Business Metrics
    runs-on: ubuntu-latest

    outputs:
      metrics_timestamp: ${{ steps.collect.outputs.timestamp }}
      revenue_total: ${{ steps.collect.outputs.revenue_total }}
      users_active: ${{ steps.collect.outputs.users_active }}

    steps:
      - name: 📥 Checkout Repository
        uses: actions/checkout@v4

      - name: 🐍 Setup Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.10'

      - name: 📦 Install Analytics Dependencies
        run: |
          pip install pandas numpy matplotlib seaborn plotly dash
          pip install supabase python-dotenv requests

      - name: 📊 Collect OASIS v3 Metrics
        id: collect
        run: |
          echo "📊 Collecting comprehensive business metrics..."

          TIMESTAMP=$(date -Iseconds)
          echo "timestamp=$TIMESTAMP" >> $GITHUB_OUTPUT

          # Collect metrics from all sources
          python scripts/analytics/collect_revenue_metrics.py             --supabase-url "$SUPABASE_URL"             --service-role-key "$SUPABASE_SERVICE_ROLE_KEY"             --output-file raw_metrics.json             --timestamp "$TIMESTAMP"

          # Extract key metrics for job outputs
          REVENUE_TOTAL=$(python -c "
          import json
          with open('raw_metrics.json') as f:
              data = json.load(f)
          print(data.get('revenue_total', 0))
          ")

          USERS_ACTIVE=$(python -c "
          import json  
          with open('raw_metrics.json') as f:
              data = json.load(f)
          print(data.get('users_active_24h', 0))
          ")

          echo "revenue_total=$REVENUE_TOTAL" >> $GITHUB_OUTPUT
          echo "users_active=$USERS_ACTIVE" >> $GITHUB_OUTPUT

      - name: 🏥 Health Check All Services
        run: |
          echo "🏥 Checking health of all OASIS v3 services..."

          python scripts/analytics/health_check_services.py             --services-config config/services.json             --output-file services_health.json             --include-response-times

      - name: 📈 HuggingFace Spaces Usage Analytics
        run: |
          echo "📈 Analyzing HuggingFace Spaces usage patterns..."

          python scripts/analytics/analyze_hf_usage.py             --spaces-list config/hf_spaces_list.json             --hf-token "${{ secrets.HF_TOKEN }}"             --output-file hf_usage_analytics.json

      - name: 📱 Termux Orchestrator Analytics
        run: |
          echo "📱 Collecting Termux orchestrator usage data..."

          python scripts/analytics/termux_usage_analytics.py             --log-sources-config config/termux_log_sources.json             --output-file termux_usage.json

      - name: 💾 Store Metrics in Database
        run: |
          python scripts/analytics/store_metrics.py             --metrics-files "raw_metrics.json,services_health.json,hf_usage_analytics.json,termux_usage.json"             --supabase-url "$SUPABASE_URL"             --service-role-key "$SUPABASE_SERVICE_ROLE_KEY"

      - name: 📤 Upload Raw Metrics
        uses: actions/upload-artifact@v3
        with:
          name: raw-metrics-${{ steps.collect.outputs.timestamp }}
          path: |
            raw_metrics.json
            services_health.json
            hf_usage_analytics.json
            termux_usage.json

  revenue_analysis:
    name: 💰 Revenue Analysis & Forecasting
    needs: collect_metrics
    runs-on: ubuntu-latest

    steps:
      - name: 📥 Checkout Repository
        uses: actions/checkout@v4

      - name: 🐍 Setup Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.10'

      - name: 📦 Install Analytics Dependencies
        run: |
          pip install pandas numpy scikit-learn matplotlib seaborn plotly
          pip install supabase prophet statsmodels

      - name: 💾 Download Metrics
        uses: actions/download-artifact@v3
        with:
          name: raw-metrics-${{ needs.collect_metrics.outputs.metrics_timestamp }}

      - name: 💰 Revenue Stream Analysis
        run: |
          echo "💰 Analyzing revenue streams and patterns..."

          python scripts/analytics/analyze_revenue_streams.py             --metrics-file raw_metrics.json             --output-file revenue_analysis.json             --include-trends             --include-forecasting

      - name: 📈 Growth Rate Calculation
        run: |
          python scripts/analytics/calculate_growth_rates.py             --supabase-url "$SUPABASE_URL"             --service-role-key "$SUPABASE_SERVICE_ROLE_KEY"             --period-days 30             --output-file growth_metrics.json

      - name: 🎯 Target Achievement Analysis
        run: |
          echo "🎯 Analyzing target achievement ($50K+/month goal)..."

          python scripts/analytics/target_achievement_analysis.py             --revenue-data revenue_analysis.json             --monthly-target 50000             --output-file target_analysis.json

      - name: 🔮 Revenue Forecasting
        if: github.event.inputs.include_forecasting == 'true' || github.event_name == 'schedule'
        run: |
          python scripts/analytics/revenue_forecasting.py             --historical-data-days 90             --forecast-days 30             --supabase-url "$SUPABASE_URL"             --service-role-key "$SUPABASE_SERVICE_ROLE_KEY"             --output-file revenue_forecast.json

      - name: 📊 Generate Revenue Dashboard Data
        run: |
          python scripts/analytics/generate_dashboard_data.py             --revenue-analysis revenue_analysis.json             --growth-metrics growth_metrics.json             --target-analysis target_analysis.json             --forecast-data revenue_forecast.json             --output-file dashboard_data.json

      - name: 📤 Upload Revenue Analysis
        uses: actions/upload-artifact@v3
        with:
          name: revenue-analysis-${{ needs.collect_metrics.outputs.metrics_timestamp }}
          path: |
            revenue_analysis.json
            growth_metrics.json
            target_analysis.json
            revenue_forecast.json
            dashboard_data.json
