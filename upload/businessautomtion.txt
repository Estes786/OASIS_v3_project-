class BusinessMetrics:
    """Track business metrics and revenue in ultra-lightweight JSON storage"""

    def __init__(self, data_file="data/business_metrics.json"):
        self.data_file = data_file
        os.makedirs(os.path.dirname(data_file), exist_ok=True)
        self.data = self._load_data()

    def _load_data(self):
        """Load existing data or create new structure"""
        try:
            with open(self.data_file, 'r') as f:
                return json.load(f)
        except FileNotFoundError:
            return {
                "clients": {},
                "services": {},
                "revenue": {
                    "daily": {},
                    "monthly": {},
                    "total": 0
                },
                "requests": [],
                "created": datetime.now().isoformat()
            }

    def _save_data(self):
        """Save data to JSON file"""
        with open(self.data_file, 'w') as f:
            json.dump(self.data, f, indent=2)

    def add_client(self, client_id, name, email, package="basic"):
        """Add new client"""
        self.data["clients"][client_id] = {
            "name": name,
            "email": email,
            "package": package,
            "created": datetime.now().isoformat(),
            "status": "active",
            "total_requests": 0,
            "total_revenue": 0
        }
        self._save_data()
        return True

    def log_request(self, client_id, service_type, cost, details=None):
        """Log a service request and calculate revenue"""
        request_data = {
            "id": len(self.data["requests"]) + 1,
            "client_id": client_id,
            "service": service_type,
            "cost": cost,
            "timestamp": datetime.now().isoformat(),
            "details": details or {}
        }

        self.data["requests"].append(request_data)

        # Update client stats
        if client_id in self.data["clients"]:
            self.data["clients"][client_id]["total_requests"] += 1
            self.data["clients"][client_id]["total_revenue"] += cost

        # Update service stats
        if service_type not in self.data["services"]:
            self.data["services"][service_type] = {"count": 0, "revenue": 0}

        self.data["services"][service_type]["count"] += 1
        self.data["services"][service_type]["revenue"] += cost

        # Update revenue tracking
        today = datetime.now().strftime("%Y-%m-%d")
        month = datetime.now().strftime("%Y-%m")

        if today not in self.data["revenue"]["daily"]:
            self.data["revenue"]["daily"][today] = 0
        if month not in self.data["revenue"]["monthly"]:
            self.data["revenue"]["monthly"][month] = 0

        self.data["revenue"]["daily"][today] += cost
        self.data["revenue"]["monthly"][month] += cost
        self.data["revenue"]["total"] += cost

        self._save_data()
        return request_data["id"]

    def get_revenue_summary(self, days=30):
        """Get revenue summary for specified days"""
        today = datetime.now().date()
        total_period = 0
        daily_avg = 0

        for i in range(days):
            date_key = (today - timedelta(days=i)).strftime("%Y-%m-%d")
            total_period += self.data["revenue"]["daily"].get(date_key, 0)

        daily_avg = total_period / days if days > 0 else 0

        return {
            "period_days": days,
            "period_revenue": total_period,
            "daily_average": daily_avg,
            "monthly_projection": daily_avg * 30,
            "total_all_time": self.data["revenue"]["total"],
            "active_clients": len([c for c in self.data["clients"].values() if c["status"] == "active"]),
            "total_requests": len(self.data["requests"])
        }

class ServicePricing:
    """Ultra-lightweight pricing calculator for AI services"""

    def __init__(self):
        self.pricing = {
            "content_generation": {
                "basic": 0.05,      # per 100 words
                "premium": 0.10,
                "enterprise": 0.15
            },
            "sentiment_analysis": {
                "basic": 0.01,      # per request
                "premium": 0.02,
                "enterprise": 0.03
            },
            "text_summarization": {
                "basic": 0.03,      # per document
                "premium": 0.06,
                "enterprise": 0.10
            },
            "translation": {
                "basic": 0.02,      # per 100 words
                "premium": 0.04,
                "enterprise": 0.06
            },
            "custom_project": {
                "basic": 50.00,     # flat rate
                "premium": 150.00,
                "enterprise": 500.00
            }
        }

    def calculate_cost(self, service_type, package, quantity=1):
        """Calculate cost for a service"""
        if service_type not in self.pricing:
            return 0

        base_price = self.pricing[service_type].get(package, self.pricing[service_type]["basic"])
        return base_price * quantity

    def get_service_menu(self):
        """Get formatted service menu"""
        menu = "ðŸ’° OASIS 2.0 SERVICE PRICING\n"
        menu += "=" * 40 + "\n"

        for service, prices in self.pricing.items():
            menu += f"\nðŸ“‹ {service.replace('_', ' ').title()}:\n"
            for package, price in prices.items():
                menu += f"  {package.capitalize()}: ${price:.2f}\n"

        return menu

class ClientManager:
    """Manage clients and their service packages"""

    def __init__(self, metrics):
        self.metrics = metrics
        self.pricing = ServicePricing()

    def create_client(self, name, email, package="basic"):
        """Create new client with unique ID"""
        client_id = f"client_{int(time.time())}"
        success = self.metrics.add_client(client_id, name, email, package)

        if success:
            return {
                "client_id": client_id,
                "name": name,
                "email": email,
                "package": package,
                "status": "created"
            }
        return None

    def process_service_request(self, client_id, service_type, data, package=None):
        """Process a service request and calculate billing"""

        # Get client info
        client = self.metrics.data["clients"].get(client_id)
        if not client:
            return {"error": "Client not found"}

        # Determine package and quantity
        if not package:
            package = client["package"]

        quantity = self._calculate_quantity(service_type, data)
        cost = self.pricing.calculate_cost(service_type, package, quantity)

        # Log the request
        request_id = self.metrics.log_request(
            client_id, 
            service_type, 
            cost, 
            {
                "quantity": quantity,
                "package": package,
                "data_preview": str(data)[:100]
            }
        )

        return {
            "request_id": request_id,
            "client_id": client_id,
            "service": service_type,
            "package": package,
            "quantity": quantity,
            "cost": cost,
            "status": "processed"
        }

    def _calculate_quantity(self, service_type, data):
        """Calculate quantity for billing purposes"""
        if isinstance(data, str):
            word_count = len(data.split())

            if service_type in ["content_generation", "translation"]:
                return max(1, word_count // 100)  # per 100 words
            elif service_type in ["sentiment_analysis", "text_summarization"]:
                return 1  # per request/document
            else:
                return 1

        return 1
