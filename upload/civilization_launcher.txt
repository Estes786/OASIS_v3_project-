    def check_installation(self):

            'scripts/civilization_hf_controller.py',
            'scripts/civilization_business_engine.py', 
            'scripts/oasis_orchestrator.py'
        ]

        missing_files = []
        for file_path in required_files:
            if not (self.base_dir / file_path).exists():

    def launch_component(self, component):
        """Launch specific component"""
        component_files = {
            'orchestrator': 'oasis_orchestrator.py',
            'hf_controller': 'civilization_hf_controller.py',
            'business': 'civilization_business_engine.py'
        }

        if component not in component_files:
            print(f"âŒ Unknown component: {component}")

        script_path = self.scripts_dir / component_files[component]
