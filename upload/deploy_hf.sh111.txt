# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

print_status() {
    echo -e "${BLUE}[INFO]${NC} $1"
}

print_success() {
    echo -e "${GREEN}[SUCCESS]${NC} $1"
}

print_warning() {
    echo -e "${YELLOW}[WARNING]${NC} $1"
}

print_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# Check if required files exist
print_status "Checking deployment files..."

required_files=("app.py" "requirements_hf.txt" "oasis_config.json")
for file in "${required_files[@]}"; do
    if [[ -f "$file" ]]; then
        print_success "$file found"
    else
        print_error "$file missing"
        echo "Please ensure all required files exist in the current directory"
        exit 1
    fi
done

# Get HF Space configuration
print_status "Configuring HuggingFace Space..."

read -p "Enter your HF Space name (username/space-name): " HF_SPACE_NAME
if [[ -z "$HF_SPACE_NAME" ]]; then
    print_error "HF Space name is required"
    exit 1
fi

read -s -p "Enter your HF Token: " HF_TOKEN
echo
if [[ -z "$HF_TOKEN" ]]; then
    print_error "HF Token is required"
    exit 1
fi
