#!/bin/bash
# =============================================================================
# FMAA Enterprise Multi-Cloud Deployment Script
# Automated deployment across GCP, AWS, Vercel, and Supabase
# =============================================================================

set -e

# Configuration
ENVIRONMENT=${1:-staging}
VERSION=${2:-latest}
PROJECT_NAME="fmaa-enterprise"

# Colors
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

log() {
    echo -e "${GREEN}[$(date +'%Y-%m-%d %H:%M:%S')] $1${NC}"
}

error() {
    echo -e "${RED}[ERROR] $1${NC}"
    exit 1
}

warn() {
    echo -e "${YELLOW}[WARNING] $1${NC}"
}

info() {
    echo -e "${BLUE}[INFO] $1${NC}"
}

# Banner
echo -e "${BLUE}"
cat << 'EOF'
╔═══════════════════════════════════════════════════════════════╗
║                                                               ║
║    🚀 FMAA Multi-Cloud Deployment                             ║
║                                                               ║
║    ☁️  GCP • AWS • Vercel • Supabase                          ║
║                                                               ║
╚═══════════════════════════════════════════════════════════════╝
EOF
echo -e "${NC}"

# Validate environment
validate_environment() {
    log "🔍 Validating deployment environment: $ENVIRONMENT"

    if [[ ! "$ENVIRONMENT" =~ ^(development|staging|production)$ ]]; then
        error "Invalid environment. Use: development, staging, or production"
    fi

    # Check required tools
    command -v docker >/dev/null 2>&1 || error "Docker is required"
    command -v kubectl >/dev/null 2>&1 || error "kubectl is required" 
    command -v gcloud >/dev/null 2>&1 || warn "gcloud CLI not found - GCP deployment will be skipped"
    command -v aws >/dev/null 2>&1 || warn "AWS CLI not found - AWS deployment will be skipped"

    log "✅ Environment validation completed"
}

# Build and push container
build_container() {
    log "🐳 Building container image..."

    IMAGE_TAG="${PROJECT_NAME}:${VERSION}-${ENVIRONMENT}"

    # Multi-stage build
    docker build \
        --target production \
        --build-arg ENVIRONMENT=$ENVIRONMENT \
        --tag $IMAGE_TAG \
        --tag ${PROJECT_NAME}:latest \
        .

    # Security scan
    if command -v trivy >/dev/null 2>&1; then
        log "🔍 Running security scan..."
        trivy image --exit-code 1 --severity HIGH,CRITICAL $IMAGE_TAG
    fi

    # Push to registries
    if [[ "$ENVIRONMENT" != "development" ]]; then
        docker push $IMAGE_TAG
        docker push ${PROJECT_NAME}:latest
    fi

    log "✅ Container build completed"
}

# Deploy to Google Cloud Platform
deploy_gcp() {
    if ! command -v gcloud >/dev/null 2>&1; then
        warn "Skipping GCP deployment - gcloud CLI not available"
        return
    fi

    log "☁️ Deploying to Google Cloud Platform..."

    # Configure kubectl for GKE
    gcloud container clusters get-credentials \
        fmaa-cluster \
        --zone us-central1-a \
        --project $GCP_PROJECT_ID

    # Apply Kubernetes manifests
    kubectl apply -f infrastructure/kubernetes/namespace.yaml
    kubectl apply -f infrastructure/kubernetes/secrets.yaml
    kubectl apply -f infrastructure/kubernetes/configmap.yaml
    kubectl apply -f infrastructure/kubernetes/deployment.yaml
    kubectl apply -f infrastructure/kubernetes/service.yaml
    kubectl apply -f infrastructure/kubernetes/ingress.yaml

    # Update image
    kubectl set image deployment/fmaa-bdi-agent \
        fmaa-bdi=$IMAGE_TAG \
        --namespace fmaa-$ENVIRONMENT

    # Wait for rollout
    kubectl rollout status deployment/fmaa-bdi-agent \
        --namespace fmaa-$ENVIRONMENT \
        --timeout=600s

    log "✅ GCP deployment completed"
}

# Deploy to AWS
deploy_aws() {
    if ! command -v aws >/dev/null 2>&1; then
        warn "Skipping AWS deployment - AWS CLI not available"
        return
    fi

    log "☁️ Deploying to Amazon Web Services..."

    # Update ECS service
    aws ecs update-service \
        --cluster fmaa-cluster \
        --service fmaa-service \
        --force-new-deployment \
        --region us-east-1

    # Deploy Lambda functions
    if [ -d "lambda/" ]; then
        for function in lambda/*/; do
            function_name=$(basename "$function")
            log "📦 Deploying Lambda function: $function_name"

            cd "$function"
            zip -r "../${function_name}.zip" .
            cd ..

            aws lambda update-function-code \
                --function-name $function_name \
                --zip-file fileb://${function_name}.zip \
                --region us-east-1

            rm "${function_name}.zip"
        done
    fi

    log "✅ AWS deployment completed"
}

# Deploy to Vercel
deploy_vercel() {
    if ! command -v vercel >/dev/null 2>&1; then
        warn "Skipping Vercel deployment - Vercel CLI not available"
        return
    fi

    log "⚡ Deploying to Vercel..."

    if [ -d "frontend/" ]; then
        cd frontend/

        # Build and deploy
        if [[ "$ENVIRONMENT" == "production" ]]; then
            vercel --prod --yes
        else
            vercel --yes
        fi

        cd ..
    fi

    log "✅ Vercel deployment completed"
}

# Setup Supabase
setup_supabase() {
    log "🗄️ Setting up Supabase..."

    if [ -f "supabase/migrations/" ]; then
        # Run migrations
        for migration in supabase/migrations/*.sql; do
            log "🔄 Running migration: $(basename $migration)"
            # Custom migration runner would go here
        done
    fi

    # Deploy Edge Functions
    if [ -d "supabase/functions/" ]; then
        for function in supabase/functions/*/; do
            function_name=$(basename "$function")
            log "📦 Deploying Edge Function: $function_name"
            # Supabase function deployment would go here
        done
    fi

    log "✅ Supabase setup completed"
}

# Deploy monitoring stack
deploy_monitoring() {
    log "📊 Deploying monitoring stack..."

    # Prometheus
    helm repo add prometheus-community https://prometheus-community.github.io/helm-charts
    helm repo update

    helm upgrade --install prometheus prometheus-community/kube-prometheus-stack \
        --namespace monitoring \
        --create-namespace \
        --values monitoring/prometheus-values.yaml \
        --set grafana.adminPassword=$GRAFANA_PASSWORD

    # Custom dashboards
    kubectl apply -f monitoring/grafana-dashboards/

    # Alerting rules
    kubectl apply -f monitoring/alerting-rules/

    log "✅ Monitoring stack deployed"
}

# Health checks
health_check() {
    log "🔍 Running post-deployment health checks..."

    # Wait for services to be ready
    sleep 30

    # Check endpoints
    ENDPOINTS=(
        "https://fmaa-${ENVIRONMENT}.yourdomain.com/health"
        "https://fmaa-${ENVIRONMENT}.yourdomain.com/metrics"
    )

    for endpoint in "${ENDPOINTS[@]}"; do
        if curl -f -s "$endpoint" > /dev/null; then
            log "✅ Health check passed: $endpoint"
        else
            error "❌ Health check failed: $endpoint"
        fi
    done

    log "✅ All health checks passed"
}

# Performance test
performance_test() {
    log "🚀 Running performance tests..."

    if command -v python3 >/dev/null 2>&1; then
        python3 scripts/performance_test.py \
            --target "https://fmaa-${ENVIRONMENT}.yourdomain.com" \
            --environment $ENVIRONMENT
    fi

    log "✅ Performance tests completed"
}

# Rollback function
rollback() {
    log "🔄 Rolling back deployment..."

    kubectl rollout undo deployment/fmaa-bdi-agent \
        --namespace fmaa-$ENVIRONMENT

    log "✅ Rollback completed"
}

# Cleanup function
cleanup() {
    log "🧹 Running cleanup..."

    # Remove temporary files
    rm -rf .tmp/

    # Clean up old images
    docker image prune -f

    log "✅ Cleanup completed"
}

# Main deployment function
main() {
    log "🚀 Starting FMAA Enterprise deployment to $ENVIRONMENT..."

    validate_environment
    build_container

    # Deploy to cloud platforms
    deploy_gcp
    deploy_aws  
    deploy_vercel
    setup_supabase

    # Setup monitoring
    if [[ "$ENVIRONMENT" != "development" ]]; then
        deploy_monitoring
    fi

    # Post-deployment checks
    health_check
    performance_test

    # Cleanup
    cleanup

    log "🎉 FMAA Enterprise deployment to $ENVIRONMENT completed successfully!"

    # Print access information
    cat << EOF

╔═══════════════════════════════════════════════════════════════╗
║                                                               ║
║    🎉 Deployment Complete!                                    ║
║                                                               ║
║    🌐 Application: https://fmaa-${ENVIRONMENT}.yourdomain.com        ║
║    📊 Monitoring: https://grafana-${ENVIRONMENT}.yourdomain.com      ║
║    📈 Metrics: https://fmaa-${ENVIRONMENT}.yourdomain.com/metrics    ║
║                                                               ║
╚═══════════════════════════════════════════════════════════════╝

EOF
}

# Handle errors and cleanup
trap cleanup EXIT
trap rollback ERR

# Execute main function
main "$@"
