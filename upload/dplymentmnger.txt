class ProcessManager:
    """Ultra-lightweight process management and supervision"""

    def __init__(self):
        self.processes = {}
        self.process_configs = {}
        self.monitoring = False
        self.restart_policy = {
            'max_retries': 5,
            'retry_interval': 10,
            'backoff_multiplier': 1.5
        }

    def register_service(self, name, command, config=None):
        """Register a service for management"""
        default_config = {
            'auto_restart': True,
            'max_memory': '100M',
            'working_dir': os.getcwd(),
            'environment': {},
            'health_check_url': None,
            'dependencies': []
        }

        if config:
            default_config.update(config)

        self.process_configs[name] = {
            'command': command,
            'config': default_config,
            'restart_count': 0,
            'last_restart': None,
            'status': 'stopped'
        }

        print(f"âœ… Registered service: {name}")
        return True

    def start_service(self, name):
        """Start a registered service"""
        if name not in self.process_configs:
            print(f"âŒ Service {name} not registered")
            return False

        if name in self.processes and self.processes[name].poll() is None:
            print(f"âš ï¸  Service {name} already running")
            return True

        config = self.process_configs[name]

        try:
            # Set up environment
            env = os.environ.copy()
            env.update(config['config']['environment'])

            # Start process
            process = subprocess.Popen(
                config['command'],
                shell=True,
                cwd=config['config']['working_dir'],
                env=env,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                preexec_fn=os.setsid  # Create new process group
            )

            self.processes[name] = process
            config['status'] = 'running'
            config['last_restart'] = time.time()

            print(f"ðŸš€ Started service: {name} (PID: {process.pid})")
            return True

        except Exception as e:
            print(f"âŒ Failed to start service {name}: {e}")
            config['status'] = 'failed'
            return False

    def stop_service(self, name, graceful=True):
        """Stop a running service"""
        if name not in self.processes:
            print(f"âš ï¸  Service {name} not running")
            return True

        process = self.processes[name]
        config = self.process_configs[name]

        try:
            if graceful:
                # Try graceful shutdown first
                os.killpg(os.getpgid(process.pid), signal.SIGTERM)

                # Wait up to 10 seconds for graceful shutdown
                for _ in range(10):
                    if process.poll() is not None:
                        break
                    time.sleep(1)

            # Force kill if still running
            if process.poll() is None:
                os.killpg(os.getpgid(process.pid), signal.SIGKILL)
                process.wait(timeout=5)

            del self.processes[name]
            config['status'] = 'stopped'

            print(f"ðŸ›‘ Stopped service: {name}")
            return True

        except Exception as e:
            print(f"âŒ Error stopping service {name}: {e}")
            return False

    def restart_service(self, name):
        """Restart a service"""
        print(f"ðŸ”„ Restarting service: {name}")
        self.stop_service(name)
        time.sleep(2)  # Brief pause
        return self.start_service(name)

    def get_service_status(self, name=None):
        """Get status of services"""
        if name:
            if name not in self.process_configs:
                return None

            config = self.process_configs[name]
            process = self.processes.get(name)

            status = {
                'name': name,
                'status': config['status'],
                'pid': process.pid if process and process.poll() is None else None,
                'restart_count': config['restart_count'],
                'last_restart': config['last_restart'],
                'uptime': time.time() - config['last_restart'] if config['last_restart'] else 0
            }

            return status

        # Return all services
        all_status = {}
        for service_name in self.process_configs:
            all_status[service_name] = self.get_service_status(service_name)

        return all_status

    def start_monitoring(self):
        """Start service health monitoring"""
        if self.monitoring:
            print("âš ï¸  Monitoring already running")
            return

        self.monitoring = True

        def monitor_loop():
            while self.monitoring:
                try:
                    for name, config in self.process_configs.items():
                        if not config['config']['auto_restart']:
                            continue

                        process = self.processes.get(name)

                        # Check if process died
                        if process and process.poll() is not None:
                            print(f"âš ï¸  Service {name} died, attempting restart...")

                            # Check restart limits
                            if config['restart_count'] >= self.restart_policy['max_retries']:
                                print(f"âŒ Service {name} exceeded max restarts, giving up")
                                config['status'] = 'failed'
                                continue

                            # Wait before restart (with backoff)
                            wait_time = self.restart_policy['retry_interval'] * (
                                self.restart_policy['backoff_multiplier'] ** config['restart_count']
                            )

                            print(f"â³ Waiting {wait_time:.1f}s before restart...")
                            time.sleep(wait_time)

                            # Attempt restart
                            if self.start_service(name):
                                config['restart_count'] += 1

                    time.sleep(5)  # Check every 5 seconds

                except Exception as e:
                    print(f"âŒ Monitoring error: {e}")
                    time.sleep(10)

        monitor_thread = threading.Thread(target=monitor_loop, daemon=True)
        monitor_thread.start()

        print("âœ… Service monitoring started")

    def stop_monitoring(self):
        """Stop service monitoring"""
        self.monitoring = False
        print("ðŸ›‘ Service monitoring stopped")

class ConfigurationManager:
    """Manage deployment configurations"""

    def __init__(self, config_dir='config'):
        self.config_dir = config_dir
        self.configs = {}
        self.load_configurations()

    def load_configurations(self):
        """Load all configuration files"""
        if not os.path.exists(self.config_dir):
            os.makedirs(self.config_dir, exist_ok=True)
            self.create_default_configs()

        try:
            for filename in os.listdir(self.config_dir):
                if filename.endswith('.json'):
                    config_name = filename[:-5]  # Remove .json
                    with open(os.path.join(self.config_dir, filename), 'r') as f:
                        self.configs[config_name] = json.load(f)
        except Exception as e:
            print(f"âš ï¸  Error loading configurations: {e}")

    def create_default_configs(self):
        """Create default configuration files"""
        default_configs = {
            'production': {

                        'auto_restart': True,

    def save_configuration(self, name, config):
        """Save configuration to file"""
        try:
            filepath = os.path.join(self.config_dir, f"{name}.json")
            with open(filepath, 'w') as f:
                json.dump(config, f, indent=2)
            self.configs[name] = config
            print(f"âœ… Saved configuration: {name}")
        except Exception as e:
            print(f"âŒ Error saving configuration {name}: {e}")

    def get_configuration(self, name):
        """Get configuration by name"""
        return self.configs.get(name)

    def list_configurations(self):
        """List all available configurations"""
        return list(self.configs.keys())

class ScalingManager:
    """Handle horizontal scaling of services"""

    def __init__(self, process_manager):
        self.process_manager = process_manager
        self.scaling_policies = {}
        self.instance_counter = defaultdict(int)

    def set_scaling_policy(self, service, min_instances=1, max_instances=3, 
                          cpu_threshold=70, memory_threshold=80):
        """Set scaling policy for a service"""
        self.scaling_policies[service] = {
            'min_instances': min_instances,
            'max_instances': max_instances,
            'cpu_threshold': cpu_threshold,
            'memory_threshold': memory_threshold,
            'current_instances': 0
        }

    def scale_service(self, service, target_instances):
        """Scale service to target number of instances"""
        if service not in self.scaling_policies:
            print(f"âŒ No scaling policy for service: {service}")
            return False

        policy = self.scaling_policies[service]
        current = policy['current_instances']

        # Ensure within limits
        target = max(policy['min_instances'], 
                    min(target_instances, policy['max_instances']))

        if target == current:
            return True

        if target > current:
            # Scale up
            for i in range(current, target):
                instance_name = f"{service}_instance_{i+1}"
                if service in self.process_manager.process_configs:
                    config = self.process_manager.process_configs[service]['config'].copy()
                    command = self.process_manager.process_configs[service]['command']

                    # Modify port if specified
                    if 'port' in config:
                        config['environment']['PORT'] = str(config['port'] + i)

                    self.process_manager.register_service(instance_name, command, config)
                    self.process_manager.start_service(instance_name)

        else:
            # Scale down
            for i in range(target, current):
                instance_name = f"{service}_instance_{i+1}"
                self.process_manager.stop_service(instance_name)

        policy['current_instances'] = target
        print(f"ðŸ“ˆ Scaled {service} to {target} instances")
        return True

    def auto_scale_check(self, metrics):
        """Check if auto-scaling is needed based on metrics"""
        for service, policy in self.scaling_policies.items():
            current = policy['current_instances']

            # Simple scaling logic based on CPU/memory
            cpu_usage = metrics.get('cpu', 0)
            memory_usage = metrics.get('memory', {}).get('percent', 0)

            should_scale_up = (
                cpu_usage > policy['cpu_threshold'] or 
                memory_usage > policy['memory_threshold']
            ) and current < policy['max_instances']

            should_scale_down = (
                cpu_usage < policy['cpu_threshold'] * 0.5 and 
                memory_usage < policy['memory_threshold'] * 0.5
            ) and current > policy['min_instances']

            if should_scale_up:
                self.scale_service(service, current + 1)
            elif should_scale_down:
                self.scale_service(service, current - 1)

class DeploymentManager:
    """Main deployment orchestrator"""

    def __init__(self):
        self.process_manager = ProcessManager()
        self.config_manager = ConfigurationManager()
        self.scaling_manager = ScalingManager(self.process_manager)
        self.current_deployment = None

    def deploy_configuration(self, config_name):
        """Deploy a specific configuration"""
        config = self.config_manager.get_configuration(config_name)
        if not config:
            print(f"âŒ Configuration {config_name} not found")
            return False

        print(f"ðŸš€ Starting deployment: {config_name}")

        try:
            # Stop existing deployment if any
            if self.current_deployment:
                self.stop_deployment()

            # Register services
            for service_name, service_config in config['services'].items():
                self.process_manager.register_service(
                    service_name, 
                    service_config['command'],
                    service_config
                )

                # Set up scaling policy
                scaling_config = config.get('scaling', {})
                self.scaling_manager.set_scaling_policy(
                    service_name,
                    min_instances=scaling_config.get('min_instances', 1),
                    max_instances=scaling_config.get('max_instances', 1),
                    cpu_threshold=scaling_config.get('cpu_threshold', 70),
                    memory_threshold=scaling_config.get('memory_threshold', 80)
                )

            # Start services with dependencies
            self._start_services_with_dependencies(config['services'])

            # Start monitoring
            self.process_manager.start_monitoring()

            self.current_deployment = config_name
            print(f"âœ… Deployment {config_name} completed successfully")
            return True

        except Exception as e:
            print(f"âŒ Deployment failed: {e}")
            return False

    def _start_services_with_dependencies(self, services):
        """Start services respecting dependency order"""
        started = set()
        remaining = dict(services)

        while remaining:
            progress = False

            for service_name, service_config in list(remaining.items()):
                dependencies = service_config.get('dependencies', [])

                # Check if all dependencies are started
                if all(dep in started for dep in dependencies):
                    if self.process_manager.start_service(service_name):
                        started.add(service_name)
                        del remaining[service_name]
                        progress = True

                        # Wait a bit for service to initialize
                        time.sleep(2)

            if not progress and remaining:
                # Start remaining services ignoring dependencies
                print("âš ï¸  Starting remaining services (dependency issues)")
                for service_name in remaining:
                    self.process_manager.start_service(service_name)
                break

    def stop_deployment(self):
        """Stop current deployment"""
        if not self.current_deployment:
            print("âš ï¸  No active deployment")
            return

        print(f"ðŸ›‘ Stopping deployment: {self.current_deployment}")

        # Stop monitoring
        self.process_manager.stop_monitoring()

        # Stop all services
        for service_name in self.process_manager.processes.keys():
            self.process_manager.stop_service(service_name)

        self.current_deployment = None
        print("âœ… Deployment stopped")

    def rolling_update(self, config_name):
        """Perform rolling update with zero downtime"""
        print(f"ðŸ”„ Starting rolling update to: {config_name}")

        # For simplicity in Termux environment, we'll do a quick restart
        # In production, this would be more sophisticated
        old_deployment = self.current_deployment

        if self.deploy_configuration(config_name):
            print(f"âœ… Rolling update completed: {old_deployment} -> {config_name}")
            return True
        else:
            print("âŒ Rolling update failed")
            return False

    def get_deployment_status(self):
        """Get comprehensive deployment status"""
        status = {
            'current_deployment': self.current_deployment,
            'timestamp': datetime.now().isoformat(),
            'services': self.process_manager.get_service_status(),
            'scaling_policies': dict(self.scaling_manager.scaling_policies)
        }

        return status

    def health_check(self):
        """Perform health check on deployment"""
        if not self.current_deployment:
            return {'healthy': False, 'reason': 'No active deployment'}

        service_status = self.process_manager.get_service_status()

        healthy_services = 0
        total_services = len(service_status)
        issues = []

        for name, status in service_status.items():
            if status['status'] == 'running' and status['pid']:
                healthy_services += 1
            else:
                issues.append(f"Service {name}: {status['status']}")

        health_score = (healthy_services / max(total_services, 1)) * 100

        return {
            'healthy': health_score >= 80,
            'health_score': health_score,
            'healthy_services': healthy_services,
            'total_services': total_services,
            'issues': issues
        }

                    print("âŒ Usage: update <config_name>")
