"""
Professional AI Business Assistant
Designed to generate revenue through business services
"""
import json
import datetime
from transformers import pipeline
import pandas as pd
import os

class BusinessAgent:
    def __init__(self, name="Business AI Assistant"):
        self.name = name
        self.services = []
        self.clients = {}
        self.revenue_generated = 0
        
        # Load AI models
        print("🤖 Loading AI models...")
        self.chatbot = pipeline("conversational", model="microsoft/DialoGPT-small")
        self.analyzer = pipeline("sentiment-analysis")
        
        # Business knowledge base
        self.services_offered = {
            "data_analysis": {
                "price": 50,
                "description": "Analyze business data and provide insights",
                "delivery_time": "24 hours"
            },
            "automation_setup": {
                "price": 200,
                "description": "Set up automated business processes",
                "delivery_time": "3-5 days"
            },
            "ai_consultation": {
                "price": 100,
                "description": "AI strategy consultation for businesses",
                "delivery_time": "2 hours"
            },
            "report_generation": {
                "price": 75,
                "description": "Generate automated business reports",
                "delivery_time": "Same day"
            }
        }
        
        print(f"✅ {self.name} initialized successfully!")
        self.log_activity("System initialized")
    
    def log_activity(self, activity):
        """Log all activities for business tracking"""
        timestamp = datetime.datetime.now().isoformat()
        log_entry = {
            "timestamp": timestamp,
            "activity": activity,
            "agent": self.name
        }
        
        # Ensure logs directory exists
        os.makedirs("logs", exist_ok=True)
        
        # Write to log file
        with open("logs/business_log.json", "a") as f:
            f.write(json.dumps(log_entry) + "\n")
    
    def handle_business_inquiry(self, client_message):
        """Handle business inquiries professionally"""
        try:
            # Analyze message sentiment
            sentiment = self.analyzer(client_message)[0]
            
            # Generate professional response
            if "price" in client_message.lower() or "cost" in client_message.lower():
                response = self.get_pricing_info()
            elif "service" in client_message.lower():
                response = self.get_services_info()
            else:
                # Use AI for general conversation
                ai_response = self.chatbot(client_message)
                response = f"Thank you for your inquiry. {ai_response}. Would you like to know about our AI business services?"
            
            self.log_activity(f"Handled inquiry: {client_message[:50]}...")
            return {
                "response": response,
                "sentiment": sentiment,
                "timestamp": datetime.datetime.now().isoformat()
            }
            
        except Exception as e:
            return {
                "response": "I apologize for the technical difficulty. Please contact us directly for assistance.",
                "error": str(e),
                "timestamp": datetime.datetime.now().isoformat()
            }
    
    def get_services_info(self):
        """Return formatted services information"""
        services_text = "🚀 Our AI Business Services:\n\n"
        for service, details in self.services_offered.items():
            services_text += f"📊 {service.replace('_', ' ').title()}\n"
            services_text += f"   💰 Price: ${details['price']}\n"
            services_text += f"   📝 {details['description']}\n"
            services_text += f"   ⏰ Delivery: {details['delivery_time']}\n\n"
        
        services_text += "💬 Contact us to get started with any service!"
        return services_text
    
    def get_pricing_info(self):
        """Return pricing information"""
        pricing_text = "💰 Our Competitive Pricing:\n\n"
        total_potential = sum(details['price'] for details in self.services_offered.values())
        
        for service, details in self.services_offered.items():
            pricing_text += f"• {service.replace('_', ' ').title()}: ${details['price']}\n"
        
        pricing_text += f"\n🎯 Bundle all services: ${int(total_potential * 0.8)} (20% discount!)"
        pricing_text += "\n\n📞 Contact us for custom quotes and enterprise solutions!"
        return pricing_text
    
    def create_client_proposal(self, client_name, requested_services):
        """Create a professional proposal for client"""
        proposal = {
            "client_name": client_name,
            "date": datetime.datetime.now().strftime("%Y-%m-%d"),
            "services": [],
            "total_cost": 0,
            "estimated_delivery": "5-7 business days"
        }
        
        for service in requested_services:
            if service in self.services_offered:
                service_info = self.services_offered[service]
                proposal["services"].append({
                    "name": service.replace('_', ' ').title(),
                    "price": service_info["price"],
                    "description": service_info["description"]
                })
                proposal["total_cost"] += service_info["price"]
        
        # Save proposal
        os.makedirs("clients", exist_ok=True)
        proposal_file = f"clients/{client_name.replace(' ', '_')}_proposal.json"
        with open(proposal_file, "w") as f:
            json.dump(proposal, f, indent=2)
        
        self.log_activity(f"Created proposal for {client_name}")
        return proposal
    
    def analyze_business_data(self, data_file):
        """Analyze business data and provide insights"""
        try:
            # Read data
            if data_file.endswith('.csv'):
                df = pd.read_csv(data_file)
            elif data_file.endswith('.xlsx'):
                df = pd.read_excel(data_file)
            else:
                return "Unsupported file format. Please provide CSV or Excel files."
            
            # Basic analysis
            analysis = {
                "total_rows": len(df),
                "total_columns": len(df.columns),
                "columns": list(df.columns),
                "missing_values": df.isnull().sum().to_dict(),
                "basic_stats": df.describe().to_dict() if df.select_dtypes(include='number').empty == False else "No numerical data",
                "insights": []
            }
            
            # Generate insights
            if len(df) > 100:
                analysis["insights"].append("✅ Good data volume for reliable analysis")
            else:
                analysis["insights"].append("⚠️ Small dataset - consider gathering more data")
            
            # Check for missing data
            missing_percent = (df.isnull().sum().sum() / (len(df) * len(df.columns))) * 100
            if missing_percent > 10:
                analysis["insights"].append(f"⚠️ {missing_percent:.1f}% missing data - data cleaning recommended")
            else:
                analysis["insights"].append("✅ Low missing data - good data quality")
            
            self.log_activity(f"Analyzed data file: {data_file}")
            return analysis
            
        except Exception as e:
            return f"Error analyzing data: {str(e)}"
    
    def generate_revenue_report(self):
        """Generate business revenue report"""
        report = {
            "total_revenue": self.revenue_generated,
            "services_offered": len(self.services_offered),
            "clients": len(self.clients),
            "average_deal_size": self.revenue_generated / max(len(self.clients), 1),
            "generated_date": datetime.datetime.now().isoformat()
        }
        return report

# Test the business agent
if __name__ == "__main__":
    print("🚀 Testing Business AI Agent...")
    
    agent = BusinessAgent()

    print(f"\n✅ Business Agent is ready to generate revenue!")

def handle_client_inquiry(message, client_name, history):
    """Handle client inquiries through web interface"""
    if not client_name:
        client_name = f"Client_{datetime.now().strftime('%H%M%S')}"
    
    # Process inquiry
    response = agent.handle_business_inquiry(message)
    
    # Update history
    history.append([f"{client_name}: {message}", response["response"]])
    
    # Store client info
    current_clients[client_name] = {
        "last_contact": datetime.now().isoformat(),
        "sentiment": response.get("sentiment", {}),
        "messages": history
    }
    
    return history, ""

"""
Daily Activity Tracker
Track daily progress towards first client
"""
import json
from datetime import datetime
import os

class DailyTracker:
    def __init__(self):
        self.activities_log = "daily_activities.json"
        self.load_activities()
    
    def load_activities(self):
        """Load existing activities"""
        try:
            with open(self.activities_log, "r") as f:
                self.activities = json.load(f)
        except FileNotFoundError:
            self.activities = []
    
    def log_activity(self, activity_type, description, result="pending"):
        """Log daily activity"""
        activity = {
            "date": datetime.now().strftime("%Y-%m-%d"),
            "time": datetime.now().strftime("%H:%M"),
            "type": activity_type,  # email, call, meeting, proposal, follow_up
            "description": description,
            "result": result,  # success, no_response, interested, not_interested, scheduled
            "value": 0  # potential revenue
        }
        
        self.activities.append(activity)
        self.save_activities()
        print(f"✅ Logged: {activity_type} - {description}")
    
    def save_activities(self):
        """Save activities to file"""
        with open(self.activities_log, "w") as f:
            json.dump(self.activities, f, indent=2)
    
    def get_today_summary(self):
        """Get today's activity summary"""
        today = datetime.now().strftime("%Y-%m-%d")
        today_activities = [a for a in self.activities if a["date"] == today]
        
        summary = {
            "date": today,
            "total_activities": len(today_activities),
            "emails_sent": len([a for a in today_activities if a["type"] == "email"]),
            "calls_made": len([a for a in today_activities if a["type"] == "call"]),
            "meetings_held": len([a for a in today_activities if a["type"] == "meeting"]),
            "proposals_sent": len([a for a in today_activities if a["type"] == "proposal"]),
            "interested_prospects": len([a for a in today_activities if a["result"] == "interested"]),
            "scheduled_meetings": len([a for a in today_activities if a["result"] == "scheduled"]),
        }
        
        return summary
    
    def get_week_summary(self):
        """Get weekly summary"""
        # Get last 7 days
        from datetime import timedelta
        week_activities = []

"""
Client Management System
Manage clients from prospect to payment
"""
import json
from datetime import datetime, timedelta
import os

class ClientManager:
    def __init__(self):
        self.clients_file = "clients_database.json"
        self.load_clients()
    
    def load_clients(self):
        """Load client database"""
        try:
            with open(self.clients_file, "r") as f:
                self.clients = json.load(f)
        except FileNotFoundError:
            self.clients = {}
    
    def save_clients(self):
        """Save client database"""
        with open(self.clients_file, "w") as f:
            json.dump(self.clients, f, indent=2)
    
    def add_client(self, name, contact_info, business_type, status="prospect"):
        """Add new client"""
        client_id = f"CLIENT_{len(self.clients) + 1:03d}"
        
        client = {
            "id": client_id,
            "name": name,
            "business_type": business_type,
            "contact_info": contact_info,
            "status": status,  # prospect, qualified, proposal_sent, client, completed
            "added_date": datetime.now().isoformat(),
            "projects": [],
            "total_revenue": 0,
            "notes": []
        }
        
        self.clients[client_id] = client
        self.save_clients()
        print(f"✅ Added client: {name} ({client_id})")
        return client_id
    
    def update_client_status(self, client_id, new_status, note=""):
        """Update client status"""
        if client_id in self.clients:
            old_status = self.clients[client_id]["status"]
            self.clients[client_id]["status"] = new_status
            
            # Add note about status change
            status_note = {
                "date": datetime.now().isoformat(),
                "type": "status_change",
                "content": f"Status changed from {old_status} to {new_status}. {note}",
                "automated": True
            }
            self.clients[client_id]["notes"].append(status_note)
            
            self.save_clients()
            print(f"✅ Updated {client_id} status: {old_status} → {new_status}")
        else:
            print(f"❌ Client {client_id} not found")
    
    def add_project(self, client_id, project_details):
        """Add project to client"""
        if client_id in self.clients:
            project = {
                "id": f"PROJECT_{len(self.clients[client_id]['projects']) + 1:03d}",
                "name": project_details["name"],
                "description": project_details["description"],
                "value": project_details["value"],
                "start_date": datetime.now().isoformat(),
                "estimated_completion": project_details.get("completion_date", ""),
                "status": "active",
                "deliverables": project_details.get("deliverables", []),
                "payments": []
            }
            
            self.clients[client_id]["projects"].append(project)
            self.save_clients()
            print(f"✅ Added project to {client_id}: {project['name']} (${project['value']})")
            return project["id"]
        else:
            print(f"❌ Client {client_id} not found")
            return None
    
    def record_payment(self, client_id, project_id, amount, payment_method=""):
        """Record payment from client"""
        if client_id in self.clients:
            client = self.clients[client_id]
            
            # Find project
            project = None
            for p in client["projects"]:
                if p["id"] == project_id:
                    project = p
                    break
            
            if project:
                payment = {
                    "date": datetime.now().isoformat(),
                    "amount": amount,
                    "method": payment_method,
                    "type": "project_payment"
                }
                
                project["payments"].append(payment)
                client["total_revenue"] += amount
                
                # Check if project is fully paid
                total_paid = sum([p["amount"] for p in project["payments"]])
                if total_paid >= project["value"]:
                    project["status"] = "completed"
                    print(f"🎉 Project {project_id} completed and fully paid!")
                
                self.save_clients()
                print(f"💰 Recorded payment: ${amount} from {client_id}")
                return True
            else:
                print(f"❌ Project {project_id} not found")
                return False
        else:
            print(f"❌ Client {client_id} not found")
            return False
    
    def get_revenue_summary(self):
        """Get revenue summary"""
        total_revenue = 0
        active_clients = 0
        completed_projects = 0
        pending_projects = 0
        
        for client_id, client in self.clients.items():
            total_revenue += client["total_revenue"]
            
            if client["status"] in ["client", "completed"]:
                active_clients += 1
            
            for project in client["projects"]:
                if project["status"] == "completed":
                    completed_projects += 1
                else:
                    pending_projects += 1
        
        summary = {
            "total_revenue": total_revenue,
            "total_clients": len(self.clients),
            "active_clients": active_clients,
            "completed_projects": completed_projects,
            "pending_projects": pending_projects,
            "average_client_value": total_revenue / max(active_clients, 1),
            "generated_date": datetime.now().isoformat()
        }
        
        return summary
    
    def get_client_pipeline(self):
        """Get sales pipeline overview"""
        pipeline = {
            "prospect": 0,
            "qualified": 0,
            "proposal_sent": 0,
            "client": 0,
            "completed": 0
        }
        
        pipeline_value = {
            "prospect": 0,
            "qualified": 0,  
            "proposal_sent": 0,
            "client": 0,
            "completed": 0
        }
        
        for client_id, client in self.clients.items():
            status = client["status"]
            if status in pipeline:
                pipeline[status] += 1
                
                # Calculate potential value
                potential_value = sum([p["value"] for p in client["projects"]])
                if potential_value == 0:
                    potential_value = 300  # Average project value
                
                pipeline_value[status] += potential_value
        
        return {
            "counts": pipeline,
            "values": pipeline_value,
            "total_pipeline_value": sum(pipeline_value.values())
        }
    
    def generate_client_report(self):
        """Generate comprehensive client report"""
        revenue_summary = self.get_revenue_summary()
        pipeline = self.get_client_pipeline()
        
        report = f"""

📈 SALES PIPELINE:
• Prospects: {pipeline['counts']['prospect']} (${pipeline['values']['prospect']})
• Qualified: {pipeline['counts']['qualified']} (${pipeline['values']['qualified']})
• Proposals Sent: {pipeline['counts']['proposal_sent']} (${pipeline['values']['proposal_sent']})
• Active Clients: {pipeline['counts']['client']} (${pipeline['values']['client']})
• Completed: {pipeline['counts']['completed']} (${pipeline['values']['completed']})

💼 PIPELINE VALUE: ${pipeline['total_pipeline_value']}

🎯 KEY METRICS:
• Conversion Rate: {(revenue_summary['active_clients'] / max(revenue_summary['total_clients'], 1) * 100):.1f}%
• Average Deal Size: ${revenue_summary['average_client_value']:.2f}
• Monthly Recurring Revenue: ${revenue_summary['total_revenue'] / max(1, (datetime.now() - datetime(2024, 1, 1)).days / 30):.2f}
        """
        
        return report
    
    def get_follow_up_reminders(self):
        """Get clients that need follow-up"""
        reminders = []
        today = datetime.now()
        
        for client_id, client in self.clients.items():
            # Check last contact date
            last_contact = datetime.fromisoformat(client["added_date"].replace('Z', '+00:00').split('+')[0])
            days_since_contact = (today - last_contact).days
            
            # Follow-up rules
            if client["status"] == "prospect" and days_since_contact >= 3:
                reminders.append({
                    "client_id": client_id,
                    "name": client["name"],
                    "type": "prospect_follow_up",

                    "priority": "high" if days_since_contact > 7 else "medium"
                })
            
            elif client["status"] == "proposal_sent" and days_since_contact >= 5:
                reminders.append({

                    "name": client["name"],
                    "type": "proposal_follow_up",
                    "days_overdue": days_since_contact - 5,
                    "priority": "high"
                })
        
        return reminders
