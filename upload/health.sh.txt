#!/bin/bash
# =============================================================================
# FMAA Enterprise Health Check Script
# Comprehensive health monitoring across all services
# =============================================================================

ENVIRONMENT=${1:-staging}
TIMEOUT=${2:-30}

# Health check endpoints
declare -A ENDPOINTS=(
    ["api"]="https://fmaa-${ENVIRONMENT}.yourdomain.com/health"
    ["metrics"]="https://fmaa-${ENVIRONMENT}.yourdomain.com/metrics"
    ["dashboard"]="https://grafana-${ENVIRONMENT}.yourdomain.com/api/health"
)

# Colors
RED='\033[0;31m'
GREEN='\033[0;32m' 
YELLOW='\033[1;33m'
NC='\033[0m'

log() {
    echo -e "${GREEN}[$(date +'%Y-%m-%d %H:%M:%S')] $1${NC}"
}

error() {
    echo -e "${RED}[ERROR] $1${NC}"
}

warn() {
    echo -e "${YELLOW}[WARNING] $1${NC}"
}

# Check single endpoint
check_endpoint() {
    local name=$1
    local url=$2
    local timeout=${3:-10}

    log "🔍 Checking $name: $url"

    if curl -f -s --max-time $timeout "$url" > /dev/null; then
        log "✅ $name is healthy"
        return 0
    else
        error "❌ $name is unhealthy"
        return 1
    fi
}

# Check all endpoints
check_all() {
    local failed=0

    for name in "${!ENDPOINTS[@]}"; do
        if ! check_endpoint "$name" "${ENDPOINTS[$name]}" $TIMEOUT; then
            ((failed++))
        fi
    done

    return $failed
}

# Main execution
main() {
    log "🔍 Starting FMAA Enterprise health check for $ENVIRONMENT..."

    if check_all; then
        log "🎉 All services are healthy!"
        exit 0
    else
        error "❌ Some services are unhealthy"
        exit 1
    fi
}

main "$@"
