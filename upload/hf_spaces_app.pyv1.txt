    def make_hf_request(self, model_name: str, payload: Dict) -> Dict:
        """
        Make request to Hugging Face Inference API
        Includes error handling and retry logic
        """
        headers = {"Authorization": f"Bearer {self.hf_token}"}
        api_url = f"{self.api_base}/{model_name}"

        try:
            response = requests.post(api_url, headers=headers, json=payload, timeout=30)

            # Handle model loading wait
            if response.status_code == 503:
                logger.info(f"Model {model_name} loading, waiting...")
                time.sleep(10)  # Wait for model to load
                response = requests.post(api_url, headers=headers, json=payload, timeout=30)

            if response.status_code == 200:
                return {"success": True, "data": response.json()}
            else:
                logger.error(f"API Error {response.status_code}: {response.text}")
                return {"success": False, "error": f"API Error: {response.status_code}"}

        except Exception as e:
            logger.error(f"Request failed: {str(e)}")
            return {"success": False, "error": str(e)}

# API Endpoint for Termux Orchestration
def create_api_endpoint():
    """
    Create API endpoint for Termux orchestration
    This enables the ultra-lightweight Termux client to make requests
    """
    def api_process(request):
        """Process API requests from Termux orchestrator"""
        try:
            data = json.loads(request)
            task_type = data.get("task_type")
            text_input = data.get("text", "")
            parameters = data.get("parameters", {})

            if task_type == "sentiment":
                result, metadata = oasis_ai.process_sentiment(text_input)
            elif task_type == "emotion":
                result, metadata = oasis_ai.process_emotion(text_input)
            elif task_type == "generation":
                max_length = parameters.get("max_length", 100)
                result, metadata = oasis_ai.generate_text(text_input, max_length)
            elif task_type == "summarization":
                max_length = parameters.get("max_length", 150)
                result, metadata = oasis_ai.summarize_text(text_input, max_length)
            elif task_type == "translation":
                result, metadata = oasis_ai.translate_text(text_input)
            elif task_type == "analytics":
                result = oasis_ai.get_analytics_dashboard()
                metadata = oasis_ai.usage_stats
            else:
                return json.dumps({"error": "Invalid task_type"})

            return json.dumps({
                "success": True,
                "result": result,
                "metadata": metadata,
                "timestamp": datetime.now().isoformat()
            })

        except Exception as e:
            return json.dumps({
                "success": False,
                "error": str(e),
                "timestamp": datetime.now().isoformat()
            })

    return api_process
