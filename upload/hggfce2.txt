```python
# hf_client.py - ZERO heavy dependencies
import requests
import json
import os
from datetime import datetime

class OasisHFClient:
    def __init__(self):
        self.api_key = os.getenv('HUGGINGFACE_API_KEY')
        self.base_url = "https://api-inference.huggingface.co/models"
        self.headers = {"Authorization": f"Bearer {self.api_key}"}
    
    def generate_text(self, prompt, model="microsoft/DialoGPT-medium"):
        """Content generation service"""
        url = f"{self.base_url}/{model}"
        payload = {"inputs": prompt, "parameters": {"max_length": 500}}
        
        response = requests.post(url, headers=self.headers, json=payload)
        return self.process_response(response)
    
    def analyze_sentiment(self, text):
        """Automation service - sentiment analysis"""
        url = f"{self.base_url}/cardiffnlp/twitter-roberta-base-sentiment-latest"
        payload = {"inputs": text}
        
        response = requests.post(url, headers=self.headers, json=payload)
        return self.process_response(response)
    
    def generate_image(self, prompt):
        """Image generation service"""
        url = f"{self.base_url}/runwayml/stable-diffusion-v1-5"
        payload = {"inputs": prompt}
        
        response = requests.post(url, headers=self.headers, json=payload)
        return response.content if response.status_code == 200 else None
    
    def process_response(self, response):
        if response.status_code == 200:
            return response.json()
        else:
            return {"error": f"API Error: {response.status_code}"}

# Quick test
if __name__ == "__main__":
    client = OasisHFClient()
    result = client.generate_text("Write a professional email about AI services")
    print(json.dumps(result, indent=2))
```

### 2. Automation Service
```python
# automation_service.py
from hf_client import OasisHFClient
import csv
import json

class AutomationService:
    def __init__(self):
        self.client = OasisHFClient()
        self.service_rate = 200  # $200 per project
    
    def batch_sentiment_analysis(self, texts):
        """Analyze sentiment for multiple texts"""
        results = []
        for text in texts:
            sentiment = self.client.analyze_sentiment(text)
            results.append({"text": text, "sentiment": sentiment})
        return results
    
    def customer_feedback_analyzer(self, feedback_file):
        """Analyze customer feedback from CSV"""
        results = []
        
        # Read feedback from file
        with open(feedback_file, 'r') as file:
            reader = csv.reader(file)
            for row in reader:
                if row:  # Skip empty rows
                    sentiment = self.client.analyze_sentiment(row[0])
                    results.append({
                        "feedback": row[0],
                        "sentiment": sentiment,
                        "timestamp": row[1] if len(row) > 1 else "N/A"
                    })
        
        return results
    
    def generate_report(self, analysis_results):
        """Generate summary report"""
        positive = sum(1 for r in analysis_results if "POSITIVE" in str(r.get("sentiment", "")))
        negative = sum(1 for r in analysis_results if "NEGATIVE" in str(r.get("sentiment", "")))
        neutral = len(analysis_results) - positive - negative
        
        report = {
            "total_analyzed": len(analysis_results),
            "positive_sentiment": positive,
            "negative_sentiment": negative,
            "neutral_sentiment": neutral,
            "positive_percentage": round((positive/len(analysis_results))*100, 2)
        }
        
        return report

# CLI Interface
def run_automation_service():
    service = AutomationService()
    
    print("🤖 OASIS Automation Service")
    print("1. Sentiment Analysis ($200)")
    print("2. Customer Feedback Analysis ($200)")
    
    choice = input("Select service (1-2): ")
    
    if choice == "1":
        texts = []
        print("Enter texts to analyze (empty line to finish):")
        while True:
            text = input("> ")
            if not text:
                break
            texts.append(text)
        
        results = service.batch_sentiment_analysis(texts)
        print("\n✅ Analysis Results:")
        print(json.dumps(results, indent=2))

if __name__ == "__main__":
    run_automation_service()
```

```python
# client_manager.py
import json
import os
from datetime import datetime

class ClientManager:
    def __init__(self):
        self.clients_file = "clients/clients.json"
        self.ensure_file_exists()
    
    def ensure_file_exists(self):
        if not os.path.exists(self.clients_file):
            with open(self.clients_file, 'w') as f:
                json.dump([], f)
    
    def add_client(self, name, email, service_type, budget):
        clients = self.load_clients()
        
        new_client = {
            "id": len(clients) + 1,
            "name": name,
            "email": email,
            "service_type": service_type,
            "budget": budget,
            "status": "active",
            "created_date": datetime.now().isoformat(),
            "projects": []
        }
        
        clients.append(new_client)
        self.save_clients(clients)
        return new_client
    
    def add_project(self, client_id, project_details):
        clients = self.load_clients()
        
        for client in clients:
            if client["id"] == client_id:
                project = {
                    "id": len(client["projects"]) + 1,
                    "details": project_details,
                    "created_date": datetime.now().isoformat(),
                    "status": "completed"
                }
                client["projects"].append(project)
                break
        
        self.save_clients(clients)
    
    def load_clients(self):
        with open(self.clients_file, 'r') as f:
            return json.load(f)
    
    def save_clients(self, clients):
        with open(self.clients_file, 'w') as f:
            json.dump(clients, f, indent=2)
    
    def generate_invoice(self, client_id, amount):
        clients = self.load_clients()
        
        for client in clients:
            if client["id"] == client_id:
                invoice = {
                    "client_name": client["name"],
                    "client_email": client["email"],
                    "amount": amount,
                    "date": datetime.now().strftime("%Y-%m-%d"),
                    "invoice_number": f"OASIS-{client_id}-{datetime.now().strftime('%Y%m%d')}"
                }
                
                # Save invoice
                with open(f"clients/invoice_{invoice['invoice_number']}.json", 'w') as f:
                    json.dump(invoice, f, indent=2)
                
                return invoice
        
        return None

# CLI Interface
def run_client_manager():
    manager = ClientManager()
    
    print("👥 OASIS Client Manager")
    print("1. Add New Client")
    print("2. View Clients")
    print("3. Generate Invoice")
    
    choice = input("Select option (1-3): ")
    
    if choice == "1":
        name = input("Client name: ")
        email = input("Client email: ")
        service = input("Service type (content/automation/chatbot): ")
        budget = float(input("Budget: $"))
        
        client = manager.add_client(name, email, service, budget)
        print(f"✅ Client added: {client}")
    
    elif choice == "2":
        clients = manager.load_clients()
        print("\n📋 Current Clients:")
        for client in clients:
            print(f"ID: {client['id']} | Name: {client['name']} | Service: {client['service_type']} | Budget: ${client['budget']}")

if __name__ == "__main__":
    run_client_manager()
```
