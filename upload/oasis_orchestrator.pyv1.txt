    def load_config(self, config_path):
        """Load configuration from JSON file"""
        default_config = {
            "hf_spaces": {
                "base_url": "YOUR_HF_SPACES_URL_HERE",
                "token": "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",
                "timeout": 30
            },

                "batch_size": 10
            },
            "mobile": {
                "max_response_time": 0.5,
                "cache_enabled": True,
                "compression": True
            }
        }

        try:
            if os.path.exists(config_path):
                with open(config_path, 'r') as f:
                    loaded_config = json.load(f)
                # Merge with defaults
                for key in default_config:
                    if key in loaded_config:
                        default_config[key].update(loaded_config[key])
                    else:
                        loaded_config[key] = default_config[key]
                return loaded_config
            else:
                # Create default config file
                with open(config_path, 'w') as f:
                    json.dump(default_config, f, indent=2)
                print(f"ðŸ“ Created default config file: {config_path}")
                return default_config
        except Exception as e:
            print(f"âš ï¸ Config error: {e}. Using defaults.")
            return default_config

    def make_request(self, endpoint, data, method="POST"):
        """
        Make HTTP request using ONLY urllib (zero dependencies)
        Optimized for mobile with timeout and error handling
        """
        try:
            # Prepare request data
            json_data = json.dumps(data).encode('utf-8')

            # Create request
            url = f"{self.config['hf_spaces']['base_url']}{endpoint}"
            req = urllib.request.Request(
                url,
                data=json_data,
                method=method
            )

            # Set headers
            req.add_header('Content-Type', 'application/json')
            req.add_header('Authorization', f"Bearer {self.config['hf_spaces']['token']}")
            req.add_header('User-Agent', 'OASIS-v3-Termux/1.0')

            # Make request with timeout
            start_time = time.time()
            with urllib.request.urlopen(req, timeout=self.config['hf_spaces']['timeout']) as response:
                response_data = response.read().decode('utf-8')
                response_time = time.time() - start_time

                # Update session stats
                self.session_stats["requests_made"] += 1
                self.session_stats["total_revenue"] += self.config['business']['pricing_per_request']

                # Parse response
                result = json.loads(response_data)
                result["response_time"] = round(response_time, 3)
                result["mobile_optimized"] = response_time < self.config['mobile']['max_response_time']

                return {
                    "success": True,
                    "data": result,
                    "response_time": response_time,
                    "mobile_optimized": response_time < self.config['mobile']['max_response_time']
                }

        except urllib.error.HTTPError as e:
            self.session_stats["errors"] += 1
            error_message = e.read().decode('utf-8') if e.fp else str(e)
            return {
                "success": False,
                "error": f"HTTP {e.code}: {error_message}",
                "error_type": "http_error"
            }
        except urllib.error.URLError as e:
            self.session_stats["errors"] += 1
            return {
                "success": False,
                "error": f"URL Error: {str(e)}",
                "error_type": "url_error"
            }
        except Exception as e:
            self.session_stats["errors"] += 1
            return {
                "success": False,
                "error": f"Request failed: {str(e)}",
                "error_type": "general_error"
            }

    def process_sentiment(self, text):
        """Process sentiment analysis via HF Spaces"""
        data = {
            "task_type": "sentiment",
            "text": text,
            "parameters": {}
        }
        return self.make_request("/api/process", data)

        data = {

        data = {

        data = {

        data = {

        data = {

    def batch_process(self, tasks):
        """
        Process multiple tasks in batch for mobile efficiency
        Reduces request overhead and improves performance
        """
        print(f"ðŸ”„ Processing {len(tasks)} tasks in batch...")
        results = []
        batch_start = time.time()

        for i, task in enumerate(tasks, 1):
            print(f"  â””â”€ Task {i}/{len(tasks)}: {task['type']}")

            if task['type'] == 'sentiment':
                result = self.process_sentiment(task['text'])
            elif task['type'] == 'emotion':
                result = self.process_emotion(task['text'])
            elif task['type'] == 'generation':
                result = self.generate_text(task['text'], task.get('max_length', 100))
            elif task['type'] == 'summarization':
                result = self.summarize_text(task['text'], task.get('max_length', 150))
            elif task['type'] == 'translation':
                result = self.translate_text(task['text'])
            else:
                result = {"success": False, "error": "Unknown task type"}

            results.append({
                "task": task,
                "result": result,
                "processed_at": datetime.now().isoformat()
            })

        batch_time = time.time() - batch_start
        print(f"âœ… Batch completed in {batch_time:.2f}s")

        return {
            "batch_results": results,
            "batch_time": batch_time,
            "tasks_processed": len(tasks),
            "average_time_per_task": batch_time / len(tasks)
        }

    def print_session_stats(self):
        """Print current session statistics"""
        runtime = datetime.now().timestamp() - datetime.fromisoformat(self.session_stats["session_start"]).timestamp()

        print("\n" + "="*50)
        print("ðŸ“Š OASIS v3 Session Statistics")
        print("="*50)
        print(f"ðŸ•’ Runtime: {runtime/60:.1f} minutes")
        print(f"ðŸ“ˆ Requests Made: {self.session_stats['requests_made']}")
        print(f"âŒ Errors: {self.session_stats['errors']}")
        print(f"ðŸ’° Session Revenue: ${self.session_stats['total_revenue']:.3f}")

        if self.session_stats["requests_made"] > 0:
            success_rate = ((self.session_stats["requests_made"] - self.session_stats["errors"]) / self.session_stats["requests_made"]) * 100
            print(f"âœ… Success Rate: {success_rate:.1f}%")

        # Monthly projection
        if runtime > 0:
            requests_per_hour = (self.session_stats["requests_made"] / runtime) * 3600
            monthly_projection = (requests_per_hour * 24 * 30) * self.config['business']['pricing_per_request']
            print(f"ðŸ“… Monthly Revenue Projection: ${monthly_projection:,.2f}")
            target_percentage = (monthly_projection / self.config['business']['target_monthly_revenue']) * 100
            print(f"ðŸŽ¯ Target Achievement: {target_percentage:.1f}%")

        print("="*50)
