class RevenueEngine:
    def __init__(self):
        self.pricing_tiers = {
            "free": {"limit": 100, "rate": 0},
            "basic": {"limit": 1000, "rate": 0.02},
            "pro": {"limit": 10000, "rate": 0.015},
            "enterprise": {"limit": 100000, "rate": 0.01}
        }
    
    async def process_payment(self, user_tier: str, usage: int):
        """Calculate and process revenue"""
        tier = self.pricing_tiers[user_tier]
        revenue = usage * tier["rate"]
        return {"revenue": revenue, "tier": user_tier}
