Copy#!/usr/bin/env python3
"""
OASIS v3 Termux Ultra-Lightweight Orchestrator
Pure Python + urllib only - ZERO heavy dependencies

Mobile AI Command Center with <5MB footprint
"""

import json
import urllib.request
import urllib.parse
import urllib.error
import os
import sys
import time
from datetime import datetime

class OasisOrchestrator:
    """Ultra-lightweight orchestrator for OASIS v3 ecosystem"""
    
    def __init__(self, config_path="config/oasis.json"):
        self.version = "3.0.0-ultra"
        self.start_time = datetime.now()
        self.config = self.load_config(config_path)
        self.session_stats = {
            "requests": 0,
            "successful": 0,
            "failed": 0,
            "revenue": 0.0
        }
        
    def load_config(self, config_path):
        """Load configuration from JSON file"""
        try:
            with open(config_path, 'r') as f:
                return json.load(f)
        except Exception as e:
            print(f"⚠️ Config error: {e}")
            return self.default_config()
    
    def default_config(self):
        """Default configuration for OASIS v3"""
        return {
            "hf_spaces": {
                "ai_processor": "https://your-username-oasis-ai-processor.hf.space",
                "orchestrator": "https://your-username-oasis-orchestrator.hf.space", 
                "business": "https://your-username-oasis-business.hf.space"
            },
            "api": {
                "timeout": 30,
                "retries": 3,
                "delay": 1.0
            },
            "business": {
                "pricing": {
                    "sentiment_analysis": 0.01,
                    "content_generation": 0.05,
                    "custom_task": 0.10
                }
            }
        }
    
    def make_request(self, url, data=None, headers=None):
        """Make HTTP request using urllib"""
        try:
            if headers is None:
                headers = {'Content-Type': 'application/json'}
                
            if data:
                data = json.dumps(data).encode('utf-8')
                
            req = urllib.request.Request(url, data=data, headers=headers)
            
            with urllib.request.urlopen(req, timeout=self.config["api"]["timeout"]) as response:
                result = json.loads(response.read().decode('utf-8'))
                self.session_stats["requests"] += 1
                self.session_stats["successful"] += 1
                return {"success": True, "data": result}
                
        except Exception as e:
            self.session_stats["requests"] += 1
            self.session_stats["failed"] += 1
            return {"success": False, "error": str(e)}
    
    def process_sentiment(self, text, model="roberta"):
        """Process sentiment analysis via AI Processor"""
        url = f"{self.config['hf_spaces']['ai_processor']}/api/sentiment"
        data = {"text": text, "model": model}
        
        result = self.make_request(url, data)
        
        if result["success"]:
            # Calculate revenue
            cost = self.config["business"]["pricing"]["sentiment_analysis"]
            self.session_stats["revenue"] += cost
            print(f"💰 Revenue: +${cost:.3f}")
            
        return result
    
    def generate_content(self, prompt, max_tokens=100):
        """Generate content via AI Processor"""
        url = f"{self.config['hf_spaces']['ai_processor']}/api/generate"
        data = {"prompt": prompt, "max_tokens": max_tokens}
        
        result = self.make_request(url, data)
        
        if result["success"]:
            cost = self.config["business"]["pricing"]["content_generation"] 
            self.session_stats["revenue"] += cost
            print(f"💰 Revenue: +${cost:.3f}")
            
        return result
    
    def get_system_status(self):
        """Get overall system status"""
        status = {
            "orchestrator_version": self.version,
            "uptime": str(datetime.now() - self.start_time),
            "session_stats": self.session_stats,
            "hf_spaces_status": {}
        }
        
        # Check each HF Space
        for space_name, space_url in self.config["hf_spaces"].items():
            health_url = f"{space_url}/api/health"
            result = self.make_request(health_url)
            status["hf_spaces_status"][space_name] = "online" if result["success"] else "offline"
        
        return status
    
    def interactive_mode(self):
        """Interactive CLI mode"""
        print("🚀 OASIS v3 Termux Orchestrator")
        print("=" * 40)
        print("Commands: sentiment, generate, status, revenue, quit")
        print()
        
        while True:
            try:
                cmd = input("oasis> ").strip().lower()
                
                if cmd == "quit" or cmd == "exit":
                    break
                elif cmd == "sentiment":
                    text = input("Enter text: ")
                    result = self.process_sentiment(text)
                    if result["success"]:
                        print(f"Sentiment: {result['data']}")
                    else:
                        print(f"Error: {result['error']}")
                        
                elif cmd == "generate":
                    prompt = input("Enter prompt: ")
                    result = self.generate_content(prompt)
                    if result["success"]:
                        print(f"Generated: {result['data']}")
                    else:
                        print(f"Error: {result['error']}")
                        
                elif cmd == "status":
                    status = self.get_system_status()
                    print(json.dumps(status, indent=2))
                    
                elif cmd == "revenue":
                    print(f"💰 Session Revenue: ${self.session_stats['revenue']:.3f}")
                    print(f"📊 Requests: {self.session_stats['requests']}")
                    print(f"✅ Success Rate: {(self.session_stats['successful']/self.session_stats['requests']*100):.1f}%" if self.session_stats['requests'] > 0 else "N/A")
                    
                else:
                    print("Unknown command. Use: sentiment, generate, status, revenue, quit")
                    
            except KeyboardInterrupt:
                print("\n👋 Goodbye!")
                break
            except Exception as e:
                print(f"Error: {e}")

if __name__ == "__main__":
    orchestrator = OasisOrchestrator()
    
    if len(sys.argv) > 1 and sys.argv[1] == "--status":
        status = orchestrator.get_system_status()
        print(json.dumps(status, indent=2))
    else:
        orchestrator.interactive_mode()

Copy#!/usr/bin/env python3
"""
OASIS v3 Business Manager
Ultra-lightweight revenue tracking and client management
"""

import json
import os
from datetime import datetime, timedelta

class BusinessManager:
    """Ultra-lightweight business operations manager"""
    
    def __init__(self, data_dir="business"):
        self.data_dir = data_dir
        os.makedirs(data_dir, exist_ok=True)
        
        self.revenue_file = os.path.join(data_dir, "revenue.json")
        self.clients_file = os.path.join(data_dir, "clients.json")
        
        self.revenue_data = self.load_revenue_data()
        self.clients_data = self.load_clients_data()
    
    def load_revenue_data(self):
        """Load revenue tracking data"""
        try:
            with open(self.revenue_file, 'r') as f:
                return json.load(f)
        except:
            return {
                "daily": {},
                "monthly": {},
                "total": 0.0,
                "transactions": []
            }
    
    def load_clients_data(self):
        """Load client management data"""
        try:
            with open(self.clients_file, 'r') as f:
                return json.load(f)
        except:
            return {
                "active_clients": 0,
                "total_clients": 0,
                "client_list": []
            }
    
    def record_revenue(self, amount, service_type, client_id=None):
        """Record a revenue transaction"""
        now = datetime.now()
        today = now.strftime("%Y-%m-%d")
        month = now.strftime("%Y-%m")
        
        # Update totals
        self.revenue_data["total"] += amount
        
        # Update daily revenue
        if today not in self.revenue_data["daily"]:
            self.revenue_data["daily"][today] = 0.0
        self.revenue_data["daily"][today] += amount
        
        # Update monthly revenue
        if month not in self.revenue_data["monthly"]:
            self.revenue_data["monthly"][month] = 0.0
        self.revenue_data["monthly"][month] += amount
        
        # Record transaction
        transaction = {
            "timestamp": now.isoformat(),
            "amount": amount,
            "service": service_type,
            "client_id": client_id
        }
        self.revenue_data["transactions"].append(transaction)
        
        # Keep only last 1000 transactions
        if len(self.revenue_data["transactions"]) > 1000:
            self.revenue_data["transactions"] = self.revenue_data["transactions"][-1000:]
        
        self.save_revenue_data()
        return True
    
    def get_revenue_summary(self):
        """Get revenue summary"""
        now = datetime.now()
        today = now.strftime("%Y-%m-%d")
        month = now.strftime("%Y-%m")
        
        return {
            "total_revenue": self.revenue_data["total"],
            "today_revenue": self.revenue_data["daily"].get(today, 0.0),
            "month_revenue": self.revenue_data["monthly"].get(month, 0.0),
            "transactions_count": len(self.revenue_data["transactions"])
        }
    
    def save_revenue_data(self):
        """Save revenue data to file"""
        try:
            with open(self.revenue_file, 'w') as f:
                json.dump(self.revenue_data, f, indent=2)
        except Exception as e:
            print(f"Error saving revenue data: {e}")
    
    def save_clients_data(self):
        """Save client data to file"""
        try:
            with open(self.clients_file, 'w') as f:
                json.dump(self.clients_data, f, indent=2)
        except Exception as e:
            print(f"Error saving client data: {e}")
    
    def add_client(self, client_info):
        """Add new client"""
        client_id = f"client_{len(self.clients_data['client_list'])}"
        client_info["id"] = client_id
        client_info["created_at"] = datetime.now().isoformat()
        client_info["revenue"] = 0.0
        
        self.clients_data["client_list"].append(client_info)
        self.clients_data["total_clients"] += 1
        self.clients_data["active_clients"] += 1
        
        self.save_clients_data()
        return client_id
    
    def get_business_metrics(self):
        """Get comprehensive business metrics"""
        revenue_summary = self.get_revenue_summary()
        
        return {
            "revenue": revenue_summary,
            "clients": {
                "total": self.clients_data["total_clients"],
                "active": self.clients_data["active_clients"]
            },
            "performance": {
                "avg_transaction": revenue_summary["total_revenue"] / max(1, revenue_summary["transactions_count"]),
                "transactions_per_day": len([t for t in self.revenue_data["transactions"] 
                                           if t["timestamp"].startswith(datetime.now().strftime("%Y-%m-%d"))])
            }
        }

if __name__ == "__main__":
    # Demo usage
    bm = BusinessManager()
    
    # Record some demo transactions
    bm.record_revenue(0.01, "sentiment_analysis", "demo_client")
    bm.record_revenue(0.05, "content_generation", "demo_client")
    
    # Show metrics
    metrics = bm.get_business_metrics()
    print("📊 Business Metrics:")
    print(json.dumps(metrics, indent=2))
