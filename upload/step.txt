### **Step 3: Create Simple BDI Agent**
**File: agents/bdi_agent.py**
```python
import json
import datetime
from transformers import pipeline

class BDIAgent:
    def __init__(self, name):
        self.name = name
        self.beliefs = {}  # What I know
        self.desires = []  # What I want
        self.intentions = []  # What I plan to do
        
        # Load AI model
        self.chatbot = pipeline("conversational", 
                               model="microsoft/DialoGPT-small")
    
    def update_belief(self, key, value):
        """Update knowledge base"""
        self.beliefs[key] = {
            "value": value,
            "timestamp": str(datetime.datetime.now())
        }
        print(f"[BELIEF] {key}: {value}")
    
    def add_desire(self, goal):
        """Add new goal"""
        self.desires.append({
            "goal": goal,
            "priority": len(self.desires) + 1,
            "status": "active"
        })
        print(f"[DESIRE] New goal: {goal}")
    
    def form_intention(self, action, reason):
        """Plan specific action"""
        intention = {
            "action": action,
            "reason": reason,
            "timestamp": str(datetime.datetime.now()),
            "status": "planned"
        }
        self.intentions.append(intention)
        print(f"[INTENTION] {action} because {reason}")
    
    def think(self, input_text):
        """Main thinking process"""
        # Update beliefs
        self.update_belief("last_input", input_text)
        
        # Generate response
        response = self.chatbot(input_text)
        
        # Form new intentions based on response
        self.form_intention("respond", "user asked question")
        
        return response
    
    def status(self):
        """Show agent status"""
        return {
            "name": self.name,
            "beliefs_count": len(self.beliefs),
            "active_desires": len([d for d in self.desires if d["status"] == "active"]),
            "planned_intentions": len([i for i in self.intentions if i["status"] == "planned"])
        }

# Test the agent
if __name__ == "__main__":
    agent = BDIAgent("Assistant Alpha")
    
    # Test conversation
    response = agent.think("Hello, what can you do?")
    print(f"Response: {response}")
    
    # Show status
    print(f"Status: {agent.status()}")
```

### **Step 5: Create Agent Manager**
**File: agents/manager.py**
```python
from bdi_agent import BDIAgent
import json

class AgentManager:
    def __init__(self):
        self.agents = {}
        self.message_history = []
    
    def create_agent(self, name, role):
        """Create new agent with specific role"""
        agent = BDIAgent(name)
        
        # Set role-specific desires
        if role == "analyst":
            agent.add_desire("analyze data patterns")
            agent.add_desire("provide insights")
        elif role == "creator":
            agent.add_desire("generate content")
            agent.add_desire("be creative")
        elif role == "coordinator":
            agent.add_desire("manage other agents")
            agent.add_desire("optimize workflow")
        
        self.agents[name] = {
            "agent": agent,
            "role": role,
            "active": True
        }
        
        print(f"Created agent: {name} ({role})")
        return agent
    
    def send_message(self, from_agent, to_agent, message):
        """Send message between agents"""
        if to_agent in self.agents:
            response = self.agents[to_agent]["agent"].think(message)
            
            # Log conversation
            self.message_history.append({
                "from": from_agent,
                "to": to_agent,
                "message": message,
                "response": response,
                "timestamp": str(datetime.datetime.now())
            })
            
            return response
        return None
    
    def get_system_status(self):
        """Get status of all agents"""
        status = {}
        for name, agent_data in self.agents.items():
            status[name] = agent_data["agent"].status()
        return status

# Test multi-agent system
if __name__ == "__main__":
    manager = AgentManager()
    
    # Create specialized agents
    analyst = manager.create_agent("DataAnalyst", "analyst")
    creator = manager.create_agent("ContentCreator", "creator")
    coordinator = manager.create_agent("Coordinator", "coordinator")
    
    # Test agent communication
    response = manager.send_message("user", "DataAnalyst", "Analyze this data: [1,2,3,4,5]")
    print(f"Analyst response: {response}")
    
    # System status
    print(f"System status: {manager.get_system_status()}")
```

### **Step 8: Create Data Analysis Module**
**File: data/processor.py**
```python
import pandas as pd
import json
from datetime import datetime

class DataProcessor:
    def __init__(self):
        self.data_store = {}
    
    def load_data(self, source, data_type="json"):
        """Load data from various sources"""
        if data_type == "json":
            with open(source, 'r') as f:
                data = json.load(f)
        elif data_type == "csv":
            data = pd.read_csv(source).to_dict('records')
        
        self.data_store[source] = {
            "data": data,
            "loaded_at": str(datetime.now()),
            "type": data_type
        }
        
        return f"Loaded {len(data)} records from {source}"
    
    def analyze_data(self, source):
        """Basic data analysis"""
        if source not in self.data_store:
            return "Data source not found"
        
        data = self.data_store[source]["data"]
        
        analysis = {
            "total_records": len(data),
            "sample_data": data[:3] if data else [],
            "keys": list(data[0].keys()) if data else [],
            "analysis_time": str(datetime.now())
        }
        
        return analysis
    
    def generate_insights(self, source):
        """Generate AI insights from data"""
        analysis = self.analyze_data(source)
        
        insights = []
        if analysis["total_records"] > 0:
            insights.append(f"Dataset contains {analysis['total_records']} records")
            insights.append(f"Available fields: {', '.join(analysis['keys'])}")
            
        return {
            "insights": insights,
            "recommendations": [
                "Consider data visualization",
                "Look for patterns in the data",
                "Check for missing values"
            ]
        }

# Test data processor
if __name__ == "__main__":
    processor = DataProcessor()
    
    # Create sample data
    sample_data = [
        {"name": "User1", "age": 25, "score": 85},
        {"name": "User2", "age": 30, "score": 92},
        {"name": "User3", "age": 28, "score": 78}
    ]
    
    with open("sample.json", "w") as f:
        json.dump(sample_data, f)
    
    # Test processing
    result = processor.load_data("sample.json")
    print(result)
    
    analysis = processor.analyze_data("sample.json")
    print(f"Analysis: {analysis}")
    
    insights = processor.generate_insights("sample.json")
    print(f"Insights: {insights}")
```

### **Step 9: Create Master Controller**
**File: main.py**
```python
#!/usr/bin/env python3
import sys
import os
sys.path.append(os.path.dirname(os.path.abspath(__file__)))

from agents.manager import AgentManager
from data.processor import DataProcessor
import gradio as gr
import json

class AISystemController:
    def __init__(self):
        self.agent_manager = AgentManager()
        self.data_processor = DataProcessor()
        self.setup_default_agents()
    
    def setup_default_agents(self):
        """Setup default agents"""
        self.agent_manager.create_agent("Analyst", "analyst")
        self.agent_manager.create_agent("Creator", "creator")
        self.agent_manager.create_agent("Coordinator", "coordinator")
    
    def process_command(self, command, data=None):
        """Process user commands"""
        command = command.lower().strip()
        
        if command == "status":
            return self.agent_manager.get_system_status()
        
        elif command.startswith("chat"):
            parts = command.split(" ", 2)
            if len(parts) >= 3:
                agent_name = parts[1]
                message = parts[2]
                response = self.agent_manager.send_message("user", agent_name, message)
                return {"response": str(response)}
        
        elif command == "analyze" and data:
            # Save data temporarily
            with open("temp_data.json", "w") as f:
                json.dump(data, f)
            
            # Process with data processor
            self.data_processor.load_data("temp_data.json")
            insights = self.data_processor.generate_insights("temp_data.json")
            
            # Get AI analysis
            ai_response = self.agent_manager.send_message(
                "user", "Analyst", f"Analyze this data: {insights}"
            )
            
            return {
                "data_insights": insights,
                "ai_analysis": str(ai_response)
            }
        
        return {"error": "Unknown command"}
    
    def launch_interface(self):
        """Launch web interface"""
        def chat_interface(message, agent, history):
            command = f"chat {agent} {message}"
            result = self.process_command(command)
            
            if "response" in result:
                history.append([message, result["response"]])
            else:
                history.append([message, "Error processing request"])
            
            return history, ""
        
        def analyze_interface(data_text):
            try:
                data = json.loads(data_text)
                result = self.process_command("analyze", data)
                return json.dumps(result, indent=2)
            except:
                return "Invalid JSON data"
        
        def status_interface():
            result = self.process_command("status")
            return json.dumps(result, indent=2)
        
        # Create interface
        with gr.Blocks(title="AI System Controller") as demo:
            gr.Markdown("# 🚀 Personal AI System Controller")
            
            with gr.Tab("💬 Chat"):
                agent_select = gr.Dropdown(
                    choices=["Analyst", "Creator", "Coordinator"],
                    value="Analyst",
                    label="Select Agent"
                )
                chatbot = gr.Chatbot()
                msg = gr.Textbox(label="Message", placeholder="Type your message...")
                
                msg.submit(chat_interface, [msg, agent_select, chatbot], [chatbot, msg])
            
            with gr.Tab("📊 Data Analysis"):
                data_input = gr.Textbox(
                    label="JSON Data",
                    placeholder='[{"key": "value"}, {"key": "value2"}]',
                    lines=5
                )
                analyze_btn = gr.Button("Analyze Data")
                analysis_output = gr.Textbox(label="Analysis Result", lines=10)
                
                analyze_btn.click(analyze_interface, data_input, analysis_output)
            
            with gr.Tab("⚙️ System Status"):
                status_btn = gr.Button("Get Status")
                status_output = gr.Textbox(label="System Status", lines=8)
                
                status_btn.click(status_interface, None, status_output)
        
        demo.launch(server_name="0.0.0.0", server_port=7860, share=True)

if __name__ == "__main__":
    print("🚀 Starting Personal AI System...")
    controller = AISystemController()
    print("✅ System initialized successfully!")
    print("🌐 Launching web interface...")
    controller.launch_interface()
```
