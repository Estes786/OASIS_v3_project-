    ```python
    # command_center.py
    import requests
    import json
    import sys

    # Ganti dengan URL Hugging Face Space Anda nanti
    API_URL = "https://NAMA_SPACE_ANDA.hf.space/run/analyze"

    def send_command(reviews_text):
        print("🚀 Mengirim titah ke Pabrik Senjata di Awan...")
        try:
            response = requests.post(API_URL, json={"data": [reviews_text]})
            if response.status_code == 200:
                result = response.json()
                print("✅ Laporan intelijen diterima!")
                print(json.dumps(result['data'], indent=2))
            else:
                print(f"❌ Pabrik Senjata melaporkan error: {response.text}")
        except Exception as e:
            print(f"❌ Gagal menghubungi Pabrik Senjata: {e}")

    if __name__ == "__main__":
        if len(sys.argv) > 1:
            # Ambil ulasan dari file
            with open(sys.argv[1], 'r') as f:
                reviews = f.read()
            send_command(reviews)
        else:
            print("Gunakan: python3 command_center.py nama_file.txt")
    ```

**File 2: `app.py` (Pabrik Senjata dan Ruang Pameran Anda)**
```python
import gradio as gr
from transformers import pipeline
import pandas as pd

# Pabrik Senjata (Model AI) akan otomatis diunduh dan disiapkan oleh Hugging Face
print("🔥 Menyiapkan Pabrik Senjata Analisis Sentimen...")
sentiment_analyzer = pipeline("sentiment-analysis")
print("✅ Pabrik Senjata siap menerima titah!")

def analyze_reviews_interface(review_text):
    reviews = [line.strip() for line in review_text.split('\n') if line.strip()]
    if not reviews:
        return "Tolong masukkan setidaknya satu ulasan."
    
    # AI bekerja di sini
    results = sentiment_analyzer(reviews)
    df = pd.DataFrame(results)
    df['review'] = reviews
    
    # Mengubah format agar lebih mudah dibaca
    df_display = df[['review', 'label', 'score']]
    df_display.rename(columns={'label': 'Sentimen', 'score': 'Tingkat Keyakinan'}, inplace=True)
    
    return df_display

# Ruang Pameran (Dashboard)
demo = gr.Interface(
    fn=analyze_reviews_interface,
    inputs=gr.Textbox(lines=15, placeholder="Masukkan ulasan pelanggan di sini, satu per baris..."),
    outputs=gr.Dataframe(headers=["Ulasan", "Sentimen", "Tingkat Keyakinan"], datatype=["str", "str", "number"]),
    title="🤖 Analisis Sentimen Pelanggan (OASIS Enterprise MVP)",
    description="MESIN PERANG OASIS V2.0. Masukkan ulasan pelanggan untuk dianalisis secara instan oleh AI di Pabrik Senjata Awan.",
    allow_flagging="never"
)

# Jalankan Ruang Pameran
demo.launch()
```
