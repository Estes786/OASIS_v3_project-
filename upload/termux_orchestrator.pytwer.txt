# ===== REVENUE TRACKER =====
class RevenueTracker:
    """Local Revenue Tracking & Business Intelligence"""

    def __init__(self):
        self.session_revenue = 0.0
        self.daily_revenue = 0.0
        self.monthly_revenue = 0.0
        self.total_requests = 0
        self.session_start = datetime.now()
        self.revenue_log = []
        self.max_log_entries = 1000  # Keep memory light

    def record_transaction(self, service: str, amount: float, response_time: float = 0):
        """Record revenue transaction locally"""
        self.session_revenue += amount
        self.daily_revenue += amount
        self.monthly_revenue += amount
        self.total_requests += 1

        transaction = {
            "timestamp": datetime.now().isoformat(),
            "service": service,
            "amount": amount,
            "response_time": response_time,
            "cumulative": self.session_revenue
        }

        # Keep only recent transactions (memory optimization)
        self.revenue_log.append(transaction)
        if len(self.revenue_log) > self.max_log_entries:
            self.revenue_log = self.revenue_log[-500:]  # Keep last 500

        return transaction

    def get_stats(self) -> Dict[str, Any]:
        """Get revenue statistics"""
        session_duration = (datetime.now() - self.session_start).total_seconds()
        requests_per_hour = (self.total_requests / max(session_duration / 3600, 0.01))
        avg_revenue_per_request = self.session_revenue / max(self.total_requests, 1)

        return {
            "session_revenue": round(self.session_revenue, 3),
            "daily_revenue": round(self.daily_revenue, 2),
            "monthly_revenue": round(self.monthly_revenue, 2),
            "total_requests": self.total_requests,
            "requests_per_hour": round(requests_per_hour, 1),
            "avg_revenue_per_request": round(avg_revenue_per_request, 3),
            "session_duration_minutes": round(session_duration / 60, 1),
            "last_transactions": self.revenue_log[-5:] if self.revenue_log else []
        }

# ===== HTTP CLIENT =====
class UltraLightHTTPClient:
    """Ultra-Lightweight HTTP Client using urllib only"""

    def __init__(self, base_url: str, timeout: int = 30):
        self.base_url = base_url.rstrip('/')
        self.timeout = timeout
        self.session_headers = {
            'Content-Type': 'application/json',
            'User-Agent': 'OASIS-v3-Termux/3.0.0-production',
            'Accept': 'application/json',
            'Accept-Encoding': 'gzip, deflate'
        }

    def make_request(self, endpoint: str, data: Dict = None, method: str = 'GET') -> Dict[str, Any]:
        """Make HTTP request using urllib"""
        url = f"{self.base_url}{endpoint}"

        try:
            if method == 'GET':
                request = urllib.request.Request(url, headers=self.session_headers)
            else:  # POST
                json_data = json.dumps(data).encode('utf-8') if data else b''
                request = urllib.request.Request(url, data=json_data, headers=self.session_headers)
                request.get_method = lambda: method

            # Execute request with timeout
            with urllib.request.urlopen(request, timeout=self.timeout) as response:
                content = response.read()

                # Handle gzip compression
                if response.headers.get('Content-Encoding') == 'gzip':
                    content = gzip.decompress(content)

                # Parse JSON response
                result = json.loads(content.decode('utf-8'))

                return {
                    "success": True,
                    "data": result,
                    "status_code": response.status,
                    "response_time": 0  # Will be calculated by caller
                }

        except urllib.error.HTTPError as e:
            error_content = e.read().decode('utf-8')
            try:
                error_data = json.loads(error_content)
            except:
                error_data = {"detail": error_content}

            return {
                "success": False,
                "error": error_data,
                "status_code": e.code,
                "message": f"HTTP {e.code}: {e.reason}"
            }

        except urllib.error.URLError as e:
            return {
                "success": False,
                "error": str(e.reason),
                "status_code": 0,
                "message": f"Connection error: {e.reason}"
            }

        except Exception as e:
            return {
                "success": False,
                "error": str(e),
                "status_code": 0,
                "message": f"Request error: {e}"
            }

def main():
    """Main execution function"""
    try:
        # Initialize OASIS v3 Orchestrator
        orchestrator = OASISTermuxOrchestrator()

        # Check command line arguments
        if len(sys.argv) > 1:
            command = sys.argv[1].lower()

            if command == "health":
                result = orchestrator.get_health_status()
                print(json.dumps(result, indent=2))
            elif command == "analytics":
                result = orchestrator.get_analytics_dashboard()
                print(json.dumps(result, indent=2))
            elif command == "sentiment" and len(sys.argv) > 2:
                text = " ".join(sys.argv[2:])
                result = orchestrator.analyze_sentiment(text)
                print(json.dumps(result, indent=2))
            elif command == "generate" and len(sys.argv) > 2:
                prompt = " ".join(sys.argv[2:])
                result = orchestrator.generate_text(prompt)
                print(json.dumps(result, indent=2))
            else:
                print("Usage:")
                print("  python termux_orchestrator.py                    # Interactive mode")
                print("  python termux_orchestrator.py health             # Health check")
                print("  python termux_orchestrator.py analytics          # Analytics")
                print("  python termux_orchestrator.py sentiment <text>   # Sentiment analysis")
                print("  python termux_orchestrator.py generate <prompt>  # Text generation")
        else:
            # Interactive CLI mode
            cli = InteractiveCLI(orchestrator)
            cli.run()

    except KeyboardInterrupt:
        print("\n\nðŸ›‘ OASIS v3 terminated by user.")
    except Exception as e:
        print(f"âŒ Fatal error: {e}")
        sys.exit(1)

if __name__ == "__main__":
    main()"
