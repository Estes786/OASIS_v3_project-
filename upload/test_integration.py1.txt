"#!/usr/bin/env python3
"""
OASIS v3 Integration Test Suite
End-to-end testing of orchestrator with HF Spaces integration
"""

import sys
import json
import time
import urllib.request
import urllib.parse
import urllib.error
from typing import Dict, List, Any, Optional

class OASISIntegrationTester:
    """Integration testing for OASIS v3 orchestrator"""

    def __init__(self, config_file: str = "oasis_config.json"):
        """Initialize integration tester"""
        self.config_file = config_file
        self.orchestrator = None
        self.test_results = []
        self.passed_tests = 0
        self.failed_tests = 0

    def setup_orchestrator(self) -> bool:
        """Setup orchestrator for testing"""
        try:
            sys.path.insert(0, '.')
            from oasis_orchestrator import OASISOrchestrator

            self.orchestrator = OASISOrchestrator(self.config_file)
            print("âœ… Orchestrator setup complete")
            return True
        except Exception as e:
            print(f"âŒ Failed to setup orchestrator: {e}")
            return False

    def log_test_result(self, test_name: str, passed: bool, details: str = "", duration: float = 0):
        """Log integration test result"""
        status = "âœ… PASS" if passed else "âŒ FAIL"
        duration_str = f" ({duration:.2f}s)" if duration > 0 else ""

        result = {
            "test": test_name,
            "status": "PASS" if passed else "FAIL",
            "details": details,
            "duration": duration
        }

        self.test_results.append(result)

        if passed:
            self.passed_tests += 1
        else:
            self.failed_tests += 1

        print(f"{status} {test_name}{duration_str}")
        if details:
            print(f"    {details}")

    def test_basic_request(self) -> bool:
        """Test basic HTTP request functionality"""
        print("\nðŸ” Testing Basic HTTP Request...")

        if not self.orchestrator:
            self.log_test_result("Basic Request", False, "Orchestrator not initialized")
            return False

        try:
            start_time = time.time()

            # Test basic GET request
            url = "https://httpbin.org/get"
            response = self.orchestrator.make_request(url, method="GET")

            duration = time.time() - start_time

            if response and '"url":' in response:
                self.log_test_result("Basic GET Request", True, f"Received valid JSON response", duration)
                return True
            else:
                self.log_test_result("Basic GET Request", False, "Invalid response format", duration)
                return False

        except Exception as e:
            self.log_test_result("Basic Request", False, f"Request failed: {e}")
            return False

    def test_json_post_request(self) -> bool:
        """Test JSON POST request functionality"""
        print("\nðŸ” Testing JSON POST Request...")

        if not self.orchestrator:
            self.log_test_result("JSON POST", False, "Orchestrator not initialized")
            return False

        try:
            start_time = time.time()

            # Test JSON POST request
            url = "https://httpbin.org/post"
            test_data = {"test": "data", "timestamp": time.time()}

            response = self.orchestrator.make_request(
                url, 
                data=json.dumps(test_data).encode('utf-8'),
                method="POST",
                headers={'Content-Type': 'application/json'}
            )

            duration = time.time() - start_time

            if response and '"json":' in response and '"test":' in response:
                self.log_test_result("JSON POST Request", True, f"Successfully posted and received JSON", duration)
                return True
            else:
                self.log_test_result("JSON POST Request", False, "POST request failed or invalid response", duration)
                return False

        except Exception as e:
            self.log_test_result("JSON POST", False, f"POST request failed: {e}")
            return False

    def test_error_handling(self) -> bool:
        """Test error handling for invalid requests"""
        print("\nðŸ” Testing Error Handling...")

        if not self.orchestrator:
            self.log_test_result("Error Handling", False, "Orchestrator not initialized")
            return False

        try:
            # Test 404 error handling
            start_time = time.time()

            url = "https://httpbin.org/status/404"
            response = self.orchestrator.make_request(url, method="GET")

            duration = time.time() - start_time

            # Should handle 404 gracefully (return None or error info)
            if response is None or "404" in str(response):
                self.log_test_result("404 Error Handling", True, f"Handled 404 error gracefully", duration)

                # Test invalid URL
                try:
                    bad_response = self.orchestrator.make_request("https://invalid-domain-12345.com/", method="GET")
                    self.log_test_result("Invalid URL Handling", True, "Handled invalid URL gracefully")
                    return True
                except Exception:
                    self.log_test_result("Invalid URL Handling", True, "Properly raised exception for invalid URL")
                    return True
            else:
                self.log_test_result("Error Handling", False, "Did not handle 404 error properly", duration)
                return False

        except Exception as e:
            # Exception handling is actually good behavior for invalid URLs
            self.log_test_result("Error Handling", True, f"Properly handled exceptions: {type(e).__name__}")
            return True

    def test_sentiment_analysis_mock(self) -> bool:
        """Test sentiment analysis with mock/test data"""
        print("\nðŸ” Testing Sentiment Analysis (Mock)...")

        if not self.orchestrator:
            self.log_test_result("Sentiment Analysis Mock", False, "Orchestrator not initialized")
            return False

        try:
            start_time = time.time()

            # Test with simple text (this will likely fail with real API due to auth)
            # but we can test the method exists and handles the call properly
            test_text = "I love this amazing product!"

            try:
                result = self.orchestrator.analyze_sentiment(test_text)
                duration = time.time() - start_time

                if result:
                    self.log_test_result("Sentiment Analysis Mock", True, f"Method executed, got result", duration)
                    return True
                else:
                    self.log_test_result("Sentiment Analysis Mock", False, "Method returned None/empty result", duration)
                    return False

            except Exception as e:
                # Expected if no valid API token - test that method exists and handles gracefully
                if "analyze_sentiment" in str(e) or "401" in str(e) or "token" in str(e).lower():
                    self.log_test_result("Sentiment Analysis Mock", True, f"Method exists, needs valid API token")
                    return True
                else:
                    self.log_test_result("Sentiment Analysis Mock", False, f"Unexpected error: {e}")
                    return False

        except AttributeError:
            self.log_test_result("Sentiment Analysis Mock", False, "analyze_sentiment method not found")
            return False
        except Exception as e:
            self.log_test_result("Sentiment Analysis Mock", False, f"Test failed: {e}")
            return False

    def test_all_ai_methods(self) -> bool:
        """Test that all AI methods exist and are callable"""
        print("\nðŸ” Testing AI Method Availability...")

        if not self.orchestrator:
            self.log_test_result("AI Methods", False, "Orchestrator not initialized")
            return False

        ai_methods = [
            'analyze_sentiment',
            'translate_text',
            'summarize_text',
            'generate_text',
            'analyze_emotion'
        ]

        methods_ok = True

        for method_name in ai_methods:
            try:
                method = getattr(self.orchestrator, method_name)
                if callable(method):
                    self.log_test_result(f"Method {method_name}", True, "Available and callable")
                else:
                    self.log_test_result(f"Method {method_name}", False, "Exists but not callable")
                    methods_ok = False
            except AttributeError:
                self.log_test_result(f"Method {method_name}", False, "Method not found")
                methods_ok = False

        return methods_ok

    def test_configuration_loading(self) -> bool:
        """Test configuration loading and validation"""
        print("\nðŸ” Testing Configuration Loading...")

        if not self.orchestrator:
            self.log_test_result("Config Loading", False, "Orchestrator not initialized")
            return False

        try:
            # Check if config was loaded
            if hasattr(self.orchestrator, 'config') and self.orchestrator.config:
                config = self.orchestrator.config

                # Check required sections
                required_sections = ['hugging_face', 'performance', 'business']
                for section in required_sections:
                    if section not in config:
                        self.log_test_result("Config Validation", False, f"Missing section: {section}")
                        return False

                # Check HF config
                hf_config = config.get('hugging_face', {})
                if 'endpoints' not in hf_config:
                    self.log_test_result("Config Validation", False, "Missing HF endpoints")
                    return False

                endpoints_count = len(hf_config.get('endpoints', {}))
                self.log_test_result("Config Loading", True, f"Loaded config with {endpoints_count} endpoints")
                return True
            else:
                self.log_test_result("Config Loading", False, "Config not loaded or empty")
                return False

        except Exception as e:
            self.log_test_result("Config Loading", False, f"Config test failed: {e}")
            return False

    def test_cli_interface(self) -> bool:
        """Test CLI interface availability"""
        print("\nðŸ” Testing CLI Interface...")

        if not self.orchestrator:
            self.log_test_result("CLI Interface", False, "Orchestrator not initialized")
            return False

        try:
            # Check if run_interactive method exists
            if hasattr(self.orchestrator, 'run_interactive') and callable(self.orchestrator.run_interactive):
                self.log_test_result("CLI Interface", True, "Interactive CLI method available")
                return True
            else:
                self.log_test_result("CLI Interface", False, "run_interactive method not found")
                return False

        except Exception as e:
            self.log_test_result("CLI Interface", False, f"CLI test failed: {e}")
            return False

    def run_integration_tests(self) -> Dict[str, Any]:
        """Run complete integration test suite"""
        print("ðŸ§ª OASIS v3 Integration Test Suite")
        print("=" * 50)

        start_time = time.time()

        # Setup orchestrator
        if not self.setup_orchestrator():
            return {
                'status': 'FAILED',
                'error': 'Could not initialize orchestrator',
                'results': []
            }

        # Run all integration tests
        test_functions = [
            self.test_configuration_loading,
            self.test_all_ai_methods,
            self.test_cli_interface,
            self.test_basic_request,
            self.test_json_post_request,
            self.test_error_handling,
            self.test_sentiment_analysis_mock
        ]

        for test_func in test_functions:
            try:
                test_func()
            except Exception as e:
                test_name = test_func.__name__.replace('test_', '').replace('_', ' ').title()
                self.log_test_result(test_name, False, f"Test execution failed: {e}")

        total_time = time.time() - start_time

        # Generate summary
        total_tests = self.passed_tests + self.failed_tests
        success_rate = (self.passed_tests / total_tests * 100) if total_tests > 0 else 0

        print("\n" + "=" * 50)
        print("ðŸ§ª INTEGRATION TEST SUMMARY")
        print("=" * 50)
        print(f"âœ… Passed: {self.passed_tests}")
        print(f"âŒ Failed: {self.failed_tests}")
        print(f"ðŸ“ˆ Success Rate: {success_rate:.1f}%")
        print(f"â±ï¸  Total Time: {total_time:.2f}s")

        # Determine overall status
        if self.failed_tests == 0:
            print("ðŸŽ‰ ALL INTEGRATION TESTS PASSED!")
            overall_status = "PASSED"
        elif success_rate >= 80:
            print("âš ï¸ MOSTLY PASSED - Some integration issues detected")
            overall_status = "MOSTLY_PASSED"
        else:
            print("ðŸš¨ INTEGRATION TESTS FAILED - Critical integration issues")
            overall_status = "FAILED"

        return {
            'status': overall_status,
            'passed': self.passed_tests,
            'failed': self.failed_tests,
            'success_rate': success_rate,
            'total_time': total_time,
            'results': self.test_results
        }

def main():
    """Main integration test function"""
    import sys

    config_file = "oasis_config.json"
    if len(sys.argv) > 1:
        config_file = sys.argv[1]

    tester = OASISIntegrationTester(config_file)
    results = tester.run_integration_tests()

    # Save results
    with open('integration_results.json', 'w') as f:
        json.dump(results, f, indent=2)

    print(f"\nðŸ’¾ Results saved to integration_results.json")

    # Exit with appropriate code
    sys.exit(0 if results['status'] in ['PASSED', 'MOSTLY_PASSED'] else 1)

if __name__ == "__main__":
    main()"
