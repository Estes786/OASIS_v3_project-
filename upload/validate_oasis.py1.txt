"#!/usr/bin/env python3
"""
OASIS v3 Validation Suite
Comprehensive testing for Termux orchestrator and HF Spaces integration
"""

import sys
import json
import time
import urllib.request
import urllib.parse
import urllib.error
from typing import Dict, List, Tuple, Any

class OASISValidator:
    """Comprehensive validation suite for OASIS v3 prototype"""

    def __init__(self, config_file: str = "oasis_config.json"):
        """Initialize validator with configuration"""
        self.config_file = config_file
        self.config = self.load_config()
        self.test_results = []
        self.passed_tests = 0
        self.failed_tests = 0

    def load_config(self) -> Dict:
        """Load configuration file"""
        try:
            with open(self.config_file, 'r') as f:
                return json.load(f)
        except Exception as e:
            print(f"âŒ Failed to load config: {e}")
            return {}

    def log_test(self, test_name: str, passed: bool, details: str = "", duration: float = 0):
        """Log test result"""
        status = "âœ… PASS" if passed else "âŒ FAIL"
        duration_str = f" ({duration:.2f}s)" if duration > 0 else ""

        result = {
            "test": test_name,
            "status": "PASS" if passed else "FAIL", 
            "details": details,
            "duration": duration
        }

        self.test_results.append(result)

        if passed:
            self.passed_tests += 1
        else:
            self.failed_tests += 1

        print(f"{status} {test_name}{duration_str}")
        if details and not passed:
            print(f"    Details: {details}")

    def test_import_dependencies(self) -> bool:
        """Test that orchestrator can import without external dependencies"""
        print("\nðŸ” Testing Import Dependencies...")

        try:
            # Test standard library imports only
            import json
            import urllib.request
            import urllib.parse  
            import urllib.error
            import sys
            import os
            import time
            import re

            self.log_test("Standard Library Imports", True, "All required standard library modules available")

            # Test that forbidden libraries are NOT available (good for Termux)
            forbidden_libs = ['numpy', 'pandas', 'torch', 'transformers', 'sklearn', 'tensorflow']
            forbidden_found = []

            for lib in forbidden_libs:
                try:
                    __import__(lib)
                    forbidden_found.append(lib)
                except ImportError:
                    pass  # Good - library not available

            if forbidden_found:
                self.log_test("Forbidden Libraries Check", False, f"Found forbidden libraries: {forbidden_found}")
                return False
            else:
                self.log_test("Forbidden Libraries Check", True, "No forbidden ML libraries found (good for Termux)")
                return True

        except Exception as e:
            self.log_test("Import Dependencies", False, str(e))
            return False

    def test_orchestrator_initialization(self) -> bool:
        """Test orchestrator can initialize properly"""
        print("\nðŸ” Testing Orchestrator Initialization...")

        try:
            # Import the orchestrator
            sys.path.insert(0, '.')
            from oasis_orchestrator import OASISOrchestrator

            # Test initialization
            start_time = time.time()
            orchestrator = OASISOrchestrator(self.config_file)
            duration = time.time() - start_time

            self.log_test("Orchestrator Import", True, "Successfully imported OASISOrchestrator")
            self.log_test("Orchestrator Init", True, f"Initialized in {duration:.2f}s", duration)

            return True

        except Exception as e:
            self.log_test("Orchestrator Initialization", False, str(e))
            return False

    def test_configuration_validation(self) -> bool:
        """Test configuration file validation"""
        print("\nðŸ” Testing Configuration Validation...")

        required_keys = ['hugging_face', 'performance', 'business']
        hf_required = ['token', 'base_url', 'endpoints']

        try:
            # Check main config structure
            for key in required_keys:
                if key not in self.config:
                    self.log_test("Config Structure", False, f"Missing required key: {key}")
                    return False

            # Check HuggingFace config
            hf_config = self.config.get('hugging_face', {})
            for key in hf_required:
                if key not in hf_config:
                    self.log_test("HF Config", False, f"Missing HF key: {key}")
                    return False

            # Check endpoints structure
            endpoints = hf_config.get('endpoints', {})
            if not endpoints:
                self.log_test("Endpoints Config", False, "No endpoints defined")
                return False

            self.log_test("Configuration Validation", True, f"All required config keys present ({len(endpoints)} endpoints)")
            return True

        except Exception as e:
            self.log_test("Configuration Validation", False, str(e))
            return False

    def test_http_connectivity(self) -> bool:
        """Test basic HTTP connectivity"""
        print("\nðŸ” Testing HTTP Connectivity...")

        # Test basic internet connectivity
        test_urls = [
            ("Google DNS", "https://dns.google/resolve?name=google.com&type=A"),
            ("HuggingFace API", "https://api-inference.huggingface.co/"),
        ]

        connectivity_ok = True

        for name, url in test_urls:
            try:
                start_time = time.time()
                req = urllib.request.Request(url)
                req.add_header('User-Agent', 'OASIS-Validator/1.0')

                with urllib.request.urlopen(req, timeout=10) as response:
                    duration = time.time() - start_time
                    status_code = response.getcode()

                if status_code == 200:
                    self.log_test(f"Connectivity: {name}", True, f"HTTP {status_code}", duration)
                else:
                    self.log_test(f"Connectivity: {name}", False, f"HTTP {status_code}", duration)
                    connectivity_ok = False

            except Exception as e:
                self.log_test(f"Connectivity: {name}", False, str(e))
                connectivity_ok = False

        return connectivity_ok

    def test_api_endpoints(self) -> bool:
        """Test API endpoint accessibility"""
        print("\nðŸ” Testing API Endpoints...")

        if not self.config:
            self.log_test("API Endpoints", False, "No configuration available")
            return False

        hf_config = self.config.get('hugging_face', {})
        token = hf_config.get('token', 'hf_dummy_token_for_testing')
        endpoints = hf_config.get('endpoints', {})

        if not endpoints:
            self.log_test("API Endpoints", False, "No endpoints configured")
            return False

        endpoints_ok = True

        for service, endpoint_url in endpoints.items():
            try:
                start_time = time.time()

                # Create test request
                req = urllib.request.Request(endpoint_url)
                req.add_header('Authorization', f'Bearer {token}')
                req.add_header('Content-Type', 'application/json')
                req.add_header('User-Agent', 'OASIS-Validator/1.0')

                # Test with HEAD request to avoid quota usage
                req.get_method = lambda: 'HEAD'

                with urllib.request.urlopen(req, timeout=15) as response:
                    duration = time.time() - start_time
                    status_code = response.getcode()

                if status_code in [200, 400, 401]:  # 400/401 expected without proper payload/token
                    self.log_test(f"Endpoint: {service}", True, f"HTTP {status_code} (accessible)", duration)
                else:
                    self.log_test(f"Endpoint: {service}", False, f"HTTP {status_code}", duration)
                    endpoints_ok = False

            except urllib.error.HTTPError as e:
                if e.code in [400, 401]:  # Expected without proper token/payload
                    self.log_test(f"Endpoint: {service}", True, f"HTTP {e.code} (accessible)")
                else:
                    self.log_test(f"Endpoint: {service}", False, f"HTTP Error {e.code}")
                    endpoints_ok = False
            except Exception as e:
                self.log_test(f"Endpoint: {service}", False, str(e))
                endpoints_ok = False

        return endpoints_ok

    def test_file_structure(self) -> bool:
        """Test required file structure"""
        print("\nðŸ” Testing File Structure...")

        required_files = [
            "oasis_orchestrator.py",
            "app.py", 
            "oasis_config.json",
            "requirements.txt",
            "termux_requirements.txt"
        ]

        optional_files = [
            "termux_setup.sh",
            "hf_deploy.sh",
            "test_oasis.py"
        ]

        files_ok = True

        for filename in required_files:
            try:
                with open(filename, 'r') as f:
                    size = len(f.read())
                self.log_test(f"Required File: {filename}", True, f"Present ({size} bytes)")
            except Exception as e:
                self.log_test(f"Required File: {filename}", False, str(e))
                files_ok = False

        for filename in optional_files:
            try:
                with open(filename, 'r') as f:
                    size = len(f.read())
                self.log_test(f"Optional File: {filename}", True, f"Present ({size} bytes)")
            except Exception:
                self.log_test(f"Optional File: {filename}", False, "Missing (optional)")

        return files_ok

    def test_orchestrator_methods(self) -> bool:
        """Test orchestrator core methods"""
        print("\nðŸ” Testing Orchestrator Methods...")

        try:
            sys.path.insert(0, '.')
            from oasis_orchestrator import OASISOrchestrator

            orchestrator = OASISOrchestrator(self.config_file)

            # Test method existence
            required_methods = [
                'make_request',
                'analyze_sentiment', 
                'translate_text',
                'summarize_text',
                'generate_text',
                'analyze_emotion'
            ]

            methods_ok = True

            for method_name in required_methods:
                if hasattr(orchestrator, method_name):
                    method = getattr(orchestrator, method_name)
                    if callable(method):
                        self.log_test(f"Method: {method_name}", True, "Present and callable")
                    else:
                        self.log_test(f"Method: {method_name}", False, "Present but not callable")
                        methods_ok = False
                else:
                    self.log_test(f"Method: {method_name}", False, "Missing")
                    methods_ok = False

            return methods_ok

        except Exception as e:
            self.log_test("Orchestrator Methods", False, str(e))
            return False

    def test_performance_constraints(self) -> bool:
        """Test performance and size constraints"""
        print("\nðŸ” Testing Performance Constraints...")

        try:
            import os

            # Check orchestrator file size (should be reasonable for mobile)
            orchestrator_size = os.path.getsize('oasis_orchestrator.py')
            size_mb = orchestrator_size / (1024 * 1024)

            if size_mb < 0.5:  # Less than 500KB
                self.log_test("Orchestrator Size", True, f"{size_mb:.2f}MB (mobile-friendly)")
            else:
                self.log_test("Orchestrator Size", False, f"{size_mb:.2f}MB (too large for mobile)")
                return False

            # Test initialization speed
            sys.path.insert(0, '.')
            from oasis_orchestrator import OASISOrchestrator

            start_time = time.time()
            orchestrator = OASISOrchestrator(self.config_file)
            init_time = time.time() - start_time

            if init_time < 2.0:  # Less than 2 seconds
                self.log_test("Init Performance", True, f"{init_time:.2f}s (fast)", init_time)
            else:
                self.log_test("Init Performance", False, f"{init_time:.2f}s (slow)", init_time)
                return False

            return True

        except Exception as e:
            self.log_test("Performance Constraints", False, str(e))
            return False

    def run_full_validation(self) -> Dict[str, Any]:
        """Run complete validation suite"""
        print("ðŸš€ OASIS v3 Validation Suite")
        print("=" * 50)

        start_time = time.time()

        # Run all tests
        tests = [
            self.test_import_dependencies,
            self.test_configuration_validation,
            self.test_file_structure,
            self.test_orchestrator_initialization,
            self.test_orchestrator_methods,
            self.test_performance_constraints,
            self.test_http_connectivity,
            self.test_api_endpoints
        ]

        for test_func in tests:
            try:
                test_func()
            except Exception as e:
                test_name = test_func.__name__.replace('test_', '').replace('_', ' ').title()
                self.log_test(test_name, False, f"Test execution failed: {e}")

        total_time = time.time() - start_time

        # Generate summary
        total_tests = self.passed_tests + self.failed_tests
        success_rate = (self.passed_tests / total_tests * 100) if total_tests > 0 else 0

        print("\n" + "=" * 50)
        print("ðŸ“Š VALIDATION SUMMARY")
        print("=" * 50)
        print(f"âœ… Passed: {self.passed_tests}")
        print(f"âŒ Failed: {self.failed_tests}")
        print(f"ðŸ“ˆ Success Rate: {success_rate:.1f}%")
        print(f"â±ï¸  Total Time: {total_time:.2f}s")

        # Determine overall status
        if self.failed_tests == 0:
            print("ðŸŽ‰ ALL TESTS PASSED - OASIS v3 is ready for deployment!")
        elif success_rate >= 80:
            print("âš ï¸  MOSTLY PASSED - Minor issues detected, review failed tests")
        else:
            print("ðŸš¨ VALIDATION FAILED - Critical issues detected, fix before deployment")

        return {
            'passed': self.passed_tests,
            'failed': self.failed_tests,
            'success_rate': success_rate,
            'total_time': total_time,
            'results': self.test_results,
            'overall_status': 'PASS' if self.failed_tests == 0 else 'FAIL'
        }

def main():
    """Main validation function"""
    import sys

    config_file = "oasis_config.json"
    if len(sys.argv) > 1:
        config_file = sys.argv[1]

    validator = OASISValidator(config_file)
    results = validator.run_full_validation()

    # Exit with appropriate code
    sys.exit(0 if results['overall_status'] == 'PASS' else 1)

if __name__ == "__main__":
    main()"
